72// Get extapi.bc file path from SVF_DIR or npm installation
73static std::string getFilePath(const std::string& path)
74{
75 std::string bcFilePath;
76
77 if (path == "SVF_DIR")
78 {
79 const char* svfdir = getenv("SVF_DIR");
80 if (!svfdir)
81 return "";
82
83 bcFilePath = svfdir;
84
85 if (!bcFilePath.empty() && bcFilePath.back() != '/')
86 bcFilePath.push_back('/');
87
88 bcFilePath.append(SVF_BUILD_TYPE "-build/lib/extapi.bc");
89 }
90 else if (path == "npm root")
91 {
92 // Check npm exists before calling `npm root`.
93 // If npm is missing, this command returns empty.
94 bcFilePath = SVFUtil::getStdoutFromCommand(
95 "command -v npm >/dev/null 2>&1 && npm root"
96 );
97
98 if (bcFilePath.empty())
99 return "";
100
101 if (!bcFilePath.empty() && bcFilePath.back() != '/')
102 bcFilePath.push_back('/');
103
104 bcFilePath.append("SVF/lib/extapi.bc");
105 }
106
107 return bcFilePath;
108}

110// Get extapi.bc path
111std::string ExtAPI::getExtBcPath()
112{
113 // Try to locate extapi.bc in the following order:
114 // 1. If setExtBcPath() has been called, use that path.
115 // 2. If the command line argument -extapi=path/to/extapi.bc is provided, use that path.
116 // 3. If SVF is being developed and used directly from a build directory, try the build dir (SVF_BUILD_DIR).
117 // 4. If the SVF_DIR environment variable is set, try $SVF_DIR with the standard relative path.
118 // 5. If installed via npm, use `npm root` and append the standard relative path.
119 // 6. As a last resort, use the directory of the loaded libSVFCore.so (or .dylib) and append extapi.bc.
120
121 std::vector<std::string> candidatePaths;
122
123 // 1. Use path set by setExtBcPath()
124 if (!extBcPath.empty())
125 return extBcPath;
126
127 // 2. Use command line argument -extapi=...
128 if (setExtBcPath(Options::ExtAPIPath()))
129 {
130 return extBcPath;
131 }
132 else
133 {
134 candidatePaths.push_back(Options::ExtAPIPath());
135 }
136
137 // 3. SVF developer: try build directory (SVF_BUILD_DIR)
138 if (setExtBcPath(SVF_BUILD_DIR "/lib/extapi.bc"))
139 {
140 return extBcPath;
141 }
142 else
143 {
144 candidatePaths.push_back(SVF_BUILD_DIR "/lib/extapi.bc");
145 }
146
147 // 4. Use $SVF_DIR environment variable + standard relative path
148 if (setExtBcPath(getFilePath("SVF_DIR")))
149 {
150 return extBcPath;
151 }
152 else
153 {
154 candidatePaths.push_back(getFilePath("SVF_DIR"));
155 }
156
157 // 5. Use npm root + standard relative path (for npm installations)
158 std::string npmPath = getFilePath("npm root");
159 if (setExtBcPath(npmPath))
160 {
161 return extBcPath;
162 }
163 else
164 {
165 candidatePaths.push_back(npmPath.empty() ? "npm root unavailable" : npmPath);
166 }
167
168 // 6. Use the directory of the loaded libSVFCore.so/.dylib + extapi.bc
169 std::string soPath = SVFUtil::getCurrentSOPath();
170 if (!soPath.empty())
171 {
172 std::string dir = soPath.substr(0, soPath.find_last_of('/'));
173 std::string candidate = dir + "/extapi.bc";
174 if (setExtBcPath(candidate))
175 {
176 return extBcPath;
177 }
178 else
179 {
180 candidatePaths.push_back(candidate);
181 }
182 }
183
184 // If all candidate paths failed, print error and suggestions
185 SVFUtil::errs() << "ERROR: Failed to locate \"extapi.bc\". Tried the following candidate paths:" << std::endl;
186 for (const auto& path : candidatePaths)
187 {
188 SVFUtil::errs() << " " << path << std::endl;
189 }
190 SVFUtil::errs() << "To override the default locations for \"extapi.bc\", you can:" << std::endl
191 << "\t1. Use the command line argument \"-extapi=path/to/extapi.bc\"" << std::endl
192 << "\t2. Use the \"setExtBcPath()\" function *BEFORE* calling \"buildSVFModule()\"" << std::endl
193 << "\t3. Override the paths in \"include/SVF/Util/config.h\" (WARNING: will be overwritten "
194 << "when rebuilding SVF (generated by CMakeLists.txt))" << std::endl;
195 abort();
196}

205bool ExtAPI::hasExtFuncAnnotation(const FunObjVar *fun, const std::string &funcAnnotation)
206{
207 assert(fun && "Null FunObjVar* pointer");
208 auto it = funObjVar2Annotations.find(fun);
209 if (it != funObjVar2Annotations.end())
210 {
211 for (const std::string& annotation : it->second)
212 if (annotation.find(funcAnnotation) != std::string::npos)
213 return true;
214 }
215 return false;
216}

269// Get the position of argument which holds the new object
270s32_t ExtAPI::get_alloc_arg_pos(const FunObjVar* F)
271{
272 std::string allocArg = getExtFuncAnnotation(F, "ALLOC_HEAP_ARG");
273 assert(!allocArg.empty() && "Not an alloc call via argument or incorrect extern function annotation!");
274
275 std::string number;
276 for (char c : allocArg)
277 {
278 if (isdigit(c))
279 number.push_back(c);
280 }
281 assert(!number.empty() && "Incorrect naming convention for svf external functions(ALLOC_HEAP_ARG + number)?");
282 return std::stoi(number);
283}

290bool ExtAPI::is_ext(const FunObjVar *F)
291{
292 assert(F && "Null FunObjVar* pointer");
293 if (F->isDeclaration() || F->isIntrinsic())
294 return true;
295 else if (hasExtFuncAnnotation(F, "OVERWRITE") && getExtFuncAnnotations(F).size() == 1)
296 return false;
297 else
298 return !getExtFuncAnnotations(F).empty();
299}
