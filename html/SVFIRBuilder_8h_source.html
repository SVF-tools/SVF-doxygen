421 inline AddrStmt* addAddrWithHeapSz(NodeID src, NodeID dst, const CallBase* cs)
422 {
423 // get name of called function
424 AddrStmt* edge = addAddrEdge(src, dst);
425
426 llvm::Function* calledFunc = cs->getCalledFunction();
427 std::string functionName;
428 if (calledFunc)
429 {
430 functionName = calledFunc->getName().str();
431 }
432 else
433 {
434 SVFUtil::writeWrnMsg("not support indirect call to add AddrStmt.\n");
435 }
436 if (functionName == "malloc")
437 {
438 if (cs->arg_size() > 0)
439 {
440 const llvm::Value* val = cs->getArgOperand(0);
441 edge->addArrSize(pag->getGNode(getValueNode(val)));
442 }
443 }
444 // Check if the function called is 'calloc' and process its arguments.
445 // e.g. "%5 = call i8* @calloc(1, 8)", edge should add two SVFValue (1 and 8)
446 else if (functionName == "calloc")
447 {
448 if (cs->arg_size() > 1)
449 {
450 edge->addArrSize(
451 pag->getGNode(getValueNode(cs->getArgOperand(0))));
452 edge->addArrSize(
453 pag->getGNode(getValueNode(cs->getArgOperand(1))));
454 }
455 }
456 else
457 {
458 if (cs->arg_size() > 0)
459 {
460 const llvm::Value* val = cs->getArgOperand(0);
461 edge->addArrSize(pag->getGNode(getValueNode(val)));
462 }
463 }
464 return edge;
465 }

477 inline CopyStmt::CopyKind getCopyKind(const Value* val)
478 {
479 // COPYVAL, ZEXT, SEXT, BITCAST, FPTRUNC, FPTOUI, FPTOSI, UITOFP, SITOFP, INTTOPTR, PTRTOINT
480 if (const Instruction* inst = SVFUtil::dyn_cast<Instruction>(val))
481 {
482 switch (inst->getOpcode())
483 {
484 case Instruction::ZExt:
485 return CopyStmt::ZEXT;
486 case Instruction::SExt:
487 return CopyStmt::SEXT;
488 case Instruction::BitCast:
489 return CopyStmt::BITCAST;
490 case Instruction ::Trunc:
491 return CopyStmt::TRUNC;
492 case Instruction::FPTrunc:
493 return CopyStmt::FPTRUNC;
494 case Instruction::FPToUI:
495 return CopyStmt::FPTOUI;
496 case Instruction::FPToSI:
497 return CopyStmt::FPTOSI;
498 case Instruction::UIToFP:
499 return CopyStmt::UITOFP;
500 case Instruction::SIToFP:
501 return CopyStmt::SITOFP;
502 case Instruction::IntToPtr:
503 return CopyStmt::INTTOPTR;
504 case Instruction::PtrToInt:
505 return CopyStmt::PTRTOINT;
506 default:
507 return CopyStmt::COPYVAL;
508 }
509 }
510 assert (false && "Unknown cast inst!");
511 abort();
512 }
