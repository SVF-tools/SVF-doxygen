179 inline void addInterleavingThread(const CxtThreadStmt& tgr, NodeID tid)
180 {
181 if(threadStmtToThreadInterLeav[tgr].test_and_set(tid))
182 {
183 instToTSMap[tgr.getStmt()].insert(tgr);
184 pushToCTSWorkList(tgr);
185 }
186 }
187 inline void addInterleavingThread(const CxtThreadStmt& tgr, const CxtThreadStmt& src)
188 {
189 bool changed = threadStmtToThreadInterLeav[tgr] |= threadStmtToThreadInterLeav[src];
190 if(changed)
191 {
192 instToTSMap[tgr.getStmt()].insert(tgr);
193 pushToCTSWorkList(tgr);
194 }
195 }
196 inline void addInterleavingBits(const CxtThreadStmt& tgr, const NodeBS& bits)
197 {
198 bool changed = threadStmtToThreadInterLeav[tgr] |= bits;
199 if(changed)
200 {
201 instToTSMap[tgr.getStmt()].insert(tgr);
202 pushToCTSWorkList(tgr);
203 }
204 }
205 inline void rmInterleavingThread(const CxtThreadStmt& tgr, const NodeBS& tids, const ICFGNode* joinsite)
206 {
207 NodeBS joinedTids;
208 for(NodeBS::iterator it = tids.begin(), eit = tids.end(); it!=eit; ++it)
209 {
210 if(isMustJoin(tgr.getTid(),joinsite))
211 joinedTids.set(*it);
212 }
213 if(threadStmtToThreadInterLeav[tgr].intersectWithComplement(joinedTids))
214 {
215 pushToCTSWorkList(tgr);
216 }
217 }

430 inline ValDomain getMarkedFlag(const CxtStmt& cs)
431 {
432 CxtStmtToAliveFlagMap::const_iterator it = cxtStmtToAliveFlagMap.find(cs);
433 if(it==cxtStmtToAliveFlagMap.end())

435 cxtStmtToAliveFlagMap[cs] = Empty;

437 }

440 }
442 void markCxtStmtFlag(const CxtStmt& tgr, ValDomain flag)
443 {
444 ValDomain flag_tgr = getMarkedFlag(tgr);
445 cxtStmtToAliveFlagMap[tgr] = flag;
446 if(flag_tgr!=getMarkedFlag(tgr))
447 {
448 instToCxtStmt[tgr.getStmt()].insert(tgr);
449 pushToCTSWorkList(tgr);
450 }
451
452 }
454 void markCxtStmtFlag(const CxtStmt& tgr, const CxtStmt& src)
455 {
456 ValDomain flag_tgr = getMarkedFlag(tgr);
457 ValDomain flag_src = getMarkedFlag(src);
458 if(flag_tgr == Empty)
459 {
460 cxtStmtToAliveFlagMap[tgr] = flag_src;
461 }
462 else if(flag_tgr == TDDead)
463 {
464 if(flag_src==TDAlive)
465 cxtStmtToAliveFlagMap[tgr] = TDAlive;
466 }
467 else
468 {
470 }
471 if(flag_tgr!=getMarkedFlag(tgr))
472 {
473 instToCxtStmt[tgr.getStmt()].insert(tgr);
474 pushToCTSWorkList(tgr);
475 }
