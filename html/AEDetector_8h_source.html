254 void addBugToReporter(const AEException& e, const ICFGNode* node,
255 const BufferOverflowInfo& info)
256 {
257
258 GenericBug::EventStack eventStack;
259 SVFBugEvent sourceInstEvent(SVFBugEvent::EventType::SourceInst, node);
260 eventStack.push_back(sourceInstEvent); // Add the source instruction event to the event stack
261
262 if (eventStack.empty())
263 {
264 return; // If the event stack is empty, return early
265 }
266
267 std::string loc = eventStack.back().getEventLoc(); // Get the location of the last event in the stack
268
269 // Check if the bug at this location has already been reported
270 if (bugLoc.find(loc) != bugLoc.end())
271 {
272 return; // If the bug location is already reported, return early
273 }
274 else
275 {
276 bugLoc.insert(loc); // Otherwise, mark this location as reported
277 }
278
279 // Add the bug to the recorder with details from the event stack
280 recoder.addAbsExecBug(GenericBug::FULLBUFOVERFLOW, eventStack,
281 info.bufferSize, info.bufferSize,
282 info.accessOffset.lb().getNumeral(), info.accessOffset.ub().getNumeral());
283 nodeToBugInfo[node] = e.what(); // Record the exception information for the node
284 }
285
289 void reportBug() override
290 {
291 if (!nodeToBugInfo.empty())
292 {
293 std::cerr << "######################Buffer Overflow (" + std::to_string(nodeToBugInfo.size())
294 + " found)######################\n";
295 std::cerr << "---------------------------------------------\n";
296 for (const auto& it : nodeToBugInfo)
297 {
298 std::cerr << it.second << "\n---------------------------------------------\n";
299 }
300 }
301 }
