227 void addBugToReporter(const ICFGNode* node, const BufferOverflowInfo& info,
228 const SVFStmt* stmt = nullptr)
229 {
230
231 GenericBug::EventStack eventStack;
232 SVFBugEvent sourceInstEvent(SVFBugEvent::EventType::SourceInst, node);
233 eventStack.push_back(sourceInstEvent); // Add the source instruction event to the event stack
234
235 if (eventStack.empty())
236 {
237 return; // If the event stack is empty, return early
238 }
239
240 std::string loc = eventStack.back().getEventLoc(); // Get the location of the last event in the stack
241
242 // Check if the bug at this location has already been reported
243 if (bugLoc.find(loc) != bugLoc.end())
244 {
245 return; // If the bug location is already reported, return early
246 }
247 else
248 {
249 bugLoc.insert(loc); // Otherwise, mark this location as reported
250 }
251
252 // Add the bug to the recorder with details from the event stack
253 recoder.addAbsExecBug(GenericBug::FULLBUFOVERFLOW, eventStack,
254 info.bufferSize.lb(), info.bufferSize.ub(),
255 info.accessOffset.lb(), info.accessOffset.ub());
256 // Duplicate locations returned above without formatting the IR again.
257 nodeToBugInfo[node] = stmt ? stmt->toString() : node->toString();
258 }
259
263 void reportBug() override
264 {
265 if (!nodeToBugInfo.empty())
266 {
267 std::cerr << "######################Buffer Overflow (" + std::to_string(nodeToBugInfo.size())
268 + " found)######################\n";
269 std::cerr << "---------------------------------------------\n";
270 for (const auto& it : nodeToBugInfo)
271 {
272 std::cerr << it.second << "\n---------------------------------------------\n";
273 }
274 }
275 }
