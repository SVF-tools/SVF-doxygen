93 struct ThreadVFCandidate
94 {
95 ThreadVFCandidate(NodeID sourceNodeId, NodeID destinationNodeId)
96 : sourceNodeId(sourceNodeId), destinationNodeId(destinationNodeId)
97 {
98 }
99
100 bool operator<(const ThreadVFCandidate& other) const
101 {
102 return sourceNodeId < other.sourceNodeId ||
103 (sourceNodeId == other.sourceNodeId &&
104 destinationNodeId < other.destinationNodeId);
105 }
106
107 bool operator==(const ThreadVFCandidate& other) const
108 {
109 return sourceNodeId == other.sourceNodeId &&
110 destinationNodeId == other.destinationNodeId;
111 }
112
113 NodeID sourceNodeId;
114 NodeID destinationNodeId;
115 };
