56SparseObjGraph::SparseObjGraph(const SVFG& svfg, ICFG& graph,
57 PointerAnalysis& pta, const PAG& ir)
58 : icfg(graph), pag(ir)
59{
60 // Step 1: collect, for each base Obj, the ICFG locations of its events.
61 Map<NodeID, Set<const ICFGNode*>> eventsByBase;
62 for (auto it = svfg.begin(); it != svfg.end(); ++it)
63 {
64 const VFGNode* vfg = it->second;
65 const ICFGNode* location = vfg->getICFGNode();
66 // Only in-function SVFG nodes participate here; global entry routes
67 // are connected below.
68 if (location && location->getFun())
69 {
70 // An indirect call edge identifies the packed Obj passed to the
71 // callee: C: f(&x) -> P: callee entry. Register both C and P as
72 // events for x so intra-function routing stops and resumes there;
73 // the interprocedural C -> P route is added here.
74 for (const VFGEdge* edge : vfg->getInEdges())
75 {
76 if (const auto* call = SVFUtil::dyn_cast<CallIndSVFGEdge>(edge))
77 {
78 const ICFGNode* caller = call->getSrcNode()->getICFGNode();
79 if (caller)
80 {
81 for (NodeID obj : call->getPointsTo())
82 {
83 if (!pag.isNullOrBlkObj(obj))
84 {
85 const NodeID base = pag.getBaseObject(obj)->getId();
86 eventsByBase[base].insert(caller);
87 eventsByBase[base].insert(location);
88 addRoute(base, caller, location, nullptr);
89 }
90 }
91 }
92 }
93 }
94
95 // MRSVFGNode already supplies points-to objects. Group each by
96 // base without modifying the SVFG or strong/weak update rules.
97 const ValVar* pointer = nullptr;
98 if (const auto* memory = SVFUtil::dyn_cast<MRSVFGNode>(vfg))
99 {
100 for (NodeID obj : memory->getPointsTo())
101 if (!pag.isNullOrBlkObj(obj))
102 eventsByBase[pag.getBaseObject(obj)->getId()].insert(location);
103 }
104 else if (const auto* store = SVFUtil::dyn_cast<StoreVFGNode>(vfg))
105 pointer = store->getLHSVar();
106 else if (const auto* load = SVFUtil::dyn_cast<LoadVFGNode>(vfg))
107 pointer = load->getRHSVar();
108
109 // Resolve ordinary load/store targets with points-to analysis.
110 // If p points to a[1], record a's base so a[0] and a[1] travel
111 // through this location as one packed Obj.
112 if (pointer)
113 {
114 for (NodeID obj : pta.getPts(pointer->getId()))
115 if (!pag.isNullOrBlkObj(obj))
116 eventsByBase[pag.getBaseObject(obj)->getId()].insert(location);
117 }
118 }
119 }
120
121 // Step 2: connect callee Obj outputs to the caller return location.
122 // For C: f(&x); R: return from the call, a RetIndSVFGEdge for x gives
123 // callee-output -> R, where AE can read the computed Obj value.
124 for (auto it = icfg.begin(); it != icfg.end(); ++it)
125 {
126 if (const auto* ret = SVFUtil::dyn_cast<RetICFGNode>(it->second))
127 {
128 for (const VFGNode* target : ret->getVFGNodes())
129 {
130 for (const VFGEdge* edge : target->getInEdges())
131 {
132 if (const auto* output = SVFUtil::dyn_cast<RetIndSVFGEdge>(edge))
133 {
134 const ICFGNode* source = output->getSrcNode()->getICFGNode();
135 if (source)
136 {
137 for (NodeID obj : output->getPointsTo())
138 {
139 if (!pag.isNullOrBlkObj(obj))
140 {
141 const NodeID base = pag.getBaseObject(obj)->getId();
142 returnedGroups[ret].set(base);
143 if (source->getFun())
144 returnedGroupsByCallee[ret][source->getFun()].set(base);
145 eventsByBase[base].insert(ret);
146 if (source->getFun())
147 eventsByBase[base].insert(source);
148 addRoute(base, source, ret, nullptr);
149 }
150 }
151 }
152 }
153 }
154 }
155 }
156 }
157
158 // Index events by location and by function: a base is routed inside a
159 // function only if it has an event there (as in the per-base walk, whose
160 // sources are the base's events and the entries of their functions).
161 Map<const ICFGNode*, NodeBS> eventsAt;
162 Map<const FunObjVar*, NodeBS> funBases;
163 Set<NodeID> globalBases;
164 for (const auto& [base, events] : eventsByBase)
165 {
166 if (pag.getBaseObject(base)->isGlobalObj())
167 globalBases.insert(base);
168 for (const ICFGNode* event : events)
169 {
170 eventsAt[event].set(base);
171 funBases[event->getFun()].set(base);
172 }
173 }
174
175 // Connect global objects from global initialization to function entries.
176 if (!globalBases.empty())
177 {
178 NodeBS globals;
179 for (NodeID base : globalBases)
180 globals.set(base);
181 const ICFGNode* global = icfg.getGlobalICFGNode();
182 for (const ICFGEdge* edge : global->getOutEdges())
183 {
184 const ICFGNode* entry = edge->getDstNode();
185 if (!SVFUtil::isa<FunEntryICFGNode>(entry))
186 continue;
187 const auto bases = funBases.find(entry->getFun());
188 if (bases == funBases.end())
189 continue;
190 NodeBS label = globals & bases->second;
191 if (!label.empty())
192 addRoute(label, global, entry, nullptr);
193 }
194 }
195
196 // Step 3: connect consecutive Obj events in ICFG order, one walk per
197 // function carrying base sets.
198 for (const auto& [fun, bases] : funBases)
199 connectFunction(fun, bases, eventsAt);
200
201 finalize();
202}

233void SparseObjGraph::addRoute(const NodeBS& bases, const ICFGNode* source,
234 const ICFGNode* target,
235 const IntraCFGEdge* branch)
236{
237 // B: if (z > 0) { use(x); use(z); }. For base z the route is
238 // B -> (B, true edge) -> Uz; AE refines B's state on the true edge
239 // once and then reads every base labeled on (B, true edge) -> Uz.
240 const NodeKey from{source, nullptr};
241 const NodeKey to{target, nullptr};
242 if (branch)
243 {
244 const NodeKey via{source, branch};
245 routeLabels[ {from, via}] |= bases;
246 routeLabels[ {via, to}] |= bases;
247 }
248 else
249 routeLabels[ {from, to}] |= bases;
250}

252void SparseObjGraph::connectFunction(
253 const FunObjVar* fun, const NodeBS& funBases,
254 const Map<const ICFGNode*, NodeBS>& eventsAt)
255{
256 // A source starts a route segment for some bases: their events, the
257 // function entry, and the control boundaries they reach. Each source
258 // processes only newly added bases; per base this is exactly the per-base
259 // walk, since a base's walk does not depend on other bases.
260 doneSources.clear();
261 pendingSources.clear();
262 sourceWorklist.clear();
263
264 addSource(icfg.getFunEntryICFGNode(fun), funBases);
265 for (const auto& event : eventsAt)
266 if (event.first->getFun() == fun)
267 addSource(event.first, event.second);
268
269 while (!sourceWorklist.empty())
270 {
271 const ICFGNode* source = sourceWorklist.front();
272 sourceWorklist.pop_front();
273 NodeBS fresh = std::move(pendingSources[source]);
274 pendingSources.erase(source);
275 fresh.intersectWithComplement(doneSources[source]);
276 if (fresh.empty())
277 continue;
278 doneSources[source] |= fresh;
279
280 if (const CallICFGNode* call = SVFUtil::dyn_cast<CallICFGNode>(source))
281 {
282 connectCallBypass(call, fresh);
283 continue;
284 }
285 for (const ICFGEdge* out : source->getOutEdges())
286 {
287 const IntraCFGEdge* intra = SVFUtil::dyn_cast<IntraCFGEdge>(out);
288 if (!intra)
289 continue;
290 const IntraCFGEdge* branch = intra->getCondition() ? intra : nullptr;
291 connectSegment(fun, source, intra->getDstNode(), branch, fresh,
292 eventsAt);
293 }
294 }
295}

297void SparseObjGraph::addSource(const ICFGNode* node, const NodeBS& bases)
298{
299 NodeBS fresh = bases;
300 const auto done = doneSources.find(node);
301 if (done != doneSources.end())
302 fresh.intersectWithComplement(done->second);
303 if (fresh.empty())
304 return;
305 NodeBS& waiting = pendingSources[node];
306 if (waiting.empty())
307 sourceWorklist.push_back(node);
308 waiting |= fresh;
309}

311void SparseObjGraph::connectCallBypass(const CallICFGNode* call,
312 const NodeBS& bases)
313{
314 // C may call f_write(&x) or f_noop(&x). If a callee has no return edge
315 // for x, retain C -> R so that candidate carries the pre-call value; a
316 // candidate with a return edge uses callee-output -> R.
317 const RetICFGNode* ret = call->getRetICFGNode();
318 NodeBS returned;
319 const auto all = returnedGroups.find(ret);
320 if (all != returnedGroups.end())
321 returned = all->second;
322 const auto byCallee = returnedGroupsByCallee.find(ret);
323 for (const ICFGEdge* out : call->getOutEdges())
324 {
325 if (!SVFUtil::isa<CallCFGEdge>(out))
326 continue;
327 const NodeBS* calleeBases = nullptr;
328 if (byCallee != returnedGroupsByCallee.end())
329 {
330 const auto callee = byCallee->second.find(out->getDstNode()->getFun());
331 if (callee != byCallee->second.end())
332 calleeBases = &callee->second;
333 }
334 if (calleeBases)
335 returned &= *calleeBases;
336 else
337 returned.clear();
338 }
339
340 NodeBS bypass = bases;
341 bypass.intersectWithComplement(returned);
342 if (bypass.empty())
343 return;
344 // The return site is a control boundary: every bypassing base stops
345 // there and starts its next segment.
346 addRoute(bypass, call, ret, nullptr);
347 addSource(ret, bypass);
348}

350void SparseObjGraph::connectSegment(
351 const FunObjVar* fun, const ICFGNode* source, const ICFGNode* start,
352 const IntraCFGEdge* branch, const NodeBS& bases,
353 const Map<const ICFGNode*, NodeBS>& eventsAt)
354{
355 // Cross ordinary ICFG nodes between events, carrying the set of bases
356 // still travelling. A base stops at its next event; every base stops at
357 // a branch, call, or return.
358 Map<const ICFGNode*, NodeBS> arrived;
359 std::deque<std::pair<const ICFGNode*, NodeBS>> walk;
360 walk.emplace_back(start, bases);
361 while (!walk.empty())
362 {
363 const ICFGNode* current = walk.front().first;
364 NodeBS travelling = std::move(walk.front().second);
365 walk.pop_front();
366 if (!current || current->getFun() != fun)
367 continue;
368 NodeBS& seen = arrived[current];
369 travelling.intersectWithComplement(seen);
370 if (travelling.empty())
371 continue;
372 seen |= travelling;
373
374 if (isControlBoundary(current))
375 {
376 addRoute(travelling, source, current, branch);
377 addSource(current, travelling);
378 continue;
379 }
380 NodeBS stop = travelling & getEvents(eventsAt, current);
381 if (!stop.empty())
382 {
383 addRoute(stop, source, current, branch);
384 travelling.intersectWithComplement(stop);
385 }
386 if (travelling.empty())
387 continue;
388 for (const ICFGEdge* next : current->getOutEdges())
389 if (const IntraCFGEdge* intra = SVFUtil::dyn_cast<IntraCFGEdge>(next))
390 walk.emplace_back(intra->getDstNode(), travelling);
391 }
392}

414const NodeBS* SparseObjGraph::intern(const NodeBS& bases)
415{
416 u64_t hash = 1469598103934665603ULL;
417 for (NodeID base : bases)
418 hash = (hash ^ base) * 1099511628211ULL;
419 auto& bucket = labels[hash];
420 for (const auto& label : bucket)
421 if (*label == bases)
422 return label.get();
423 bucket.push_back(std::make_unique<NodeBS>(bases));
424 return bucket.back().get();
425}

427void SparseObjGraph::finalize()
428{
429 for (const auto& [key, bases] : routeLabels)
430 {
431 Node* source = getOrCreateNode(key.first.first, key.first.second);
432 Node* target = getOrCreateNode(key.second.first, key.second.second);
433 Edge* edge = new Edge(source, target, intern(bases));
434 const bool addedOut = source->addOutgoingEdge(edge);
435 const bool addedIn = target->addIncomingEdge(edge);
436 assert(addedOut && addedIn);
437 (void)addedOut;
438 (void)addedIn;
439 incEdgeNum();
440 if (!target->branch)
441 inputs[target->location].push_back(edge);
442 }
443 routeLabels.clear();
444}
