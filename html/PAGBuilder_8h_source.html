428 inline AddrStmt* addAddrWithHeapSz(NodeID src, NodeID dst, const CallBase* cs)
429 {
430 // get name of called function
431 AddrStmt* edge = addAddrEdge(src, dst);
432
433 llvm::Function* calledFunc = cs->getCalledFunction();
434 std::string functionName;
435 if (calledFunc)
436 {
437 functionName = calledFunc->getName().str();
438 }
439 else
440 {
441 SVFUtil::writeWrnMsg("not support indirect call to add AddrStmt.\n");
442 }
443 if (functionName == "malloc")
444 {
445 if (cs->arg_size() > 0)
446 {
447 const llvm::Value* val = cs->getArgOperand(0);
448 edge->addArrSize(pag->getGNode(getValueNode(val)));
449 }
450 }
451 // Check if the function called is 'calloc' and process its arguments.
452 // e.g. "%5 = call i8* @calloc(1, 8)", edge should add two SVFValue (1 and 8)
453 else if (functionName == "calloc")
454 {
455 if (cs->arg_size() > 1)
456 {
457 edge->addArrSize(
458 pag->getGNode(getValueNode(cs->getArgOperand(0))));
459 edge->addArrSize(
460 pag->getGNode(getValueNode(cs->getArgOperand(1))));
461 }
462 }
463 else
464 {
465 if (cs->arg_size() > 0)
466 {
467 const llvm::Value* val = cs->getArgOperand(0);
468 edge->addArrSize(pag->getGNode(getValueNode(val)));
469 }
470 }
471 return edge;
472 }

484 inline CopyStmt::CopyKind getCopyKind(const Value* val)
485 {
486 // COPYVAL, ZEXT, SEXT, BITCAST, FPTRUNC, FPTOUI, FPTOSI, UITOFP, SITOFP, INTTOPTR, PTRTOINT
487 if (const Instruction* inst = SVFUtil::dyn_cast<Instruction>(val))
488 {
489 switch (inst->getOpcode())
490 {
491 case Instruction::ZExt:
492 return CopyStmt::ZEXT;
493 case Instruction::SExt:
494 return CopyStmt::SEXT;
495 case Instruction::BitCast:
496 return CopyStmt::BITCAST;
497 case Instruction ::Trunc:
498 return CopyStmt::TRUNC;
499 case Instruction::FPTrunc:
500 return CopyStmt::FPTRUNC;
501 case Instruction::FPToUI:
502 return CopyStmt::FPTOUI;
503 case Instruction::FPToSI:
504 return CopyStmt::FPTOSI;
505 case Instruction::UIToFP:
506 return CopyStmt::UITOFP;
507 case Instruction::SIToFP:
508 return CopyStmt::SITOFP;
509 case Instruction::IntToPtr:
510 return CopyStmt::INTTOPTR;
511 case Instruction::PtrToInt:
512 return CopyStmt::PTRTOINT;
513 default:
514 return CopyStmt::COPYVAL;
515 }
516 }
517 assert (false && "Unknown cast inst!");
518 abort();
519 }
