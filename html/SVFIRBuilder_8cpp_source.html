55SVFIR* SVFIRBuilder::build()
56{
57 double startTime = SVFStat::getClk(true);
58
59 DBOUT(DGENERAL, outs() << pasMsg("\t Building SVFIR ...\n"));
60
61 // If the SVFIR has been built before, then we return the unique SVFIR of the program
62 if(pag->getNodeNumAfterPAGBuild() > 1)
63 return pag;
64
65
66 createFunObjVars();
67
69 ICFGBuilder icfgbuilder;
70 pag->icfg = icfgbuilder.build();
71
74 initialiseNodes();
75 initialiseAggregateReturns();
78 visitGlobal();
80
81
82
84 CallGraphBuilder callGraphBuilder;
85 std::vector<const FunObjVar*> funset;
86 for (const auto& item: llvmModuleSet()->getFunctionSet())
87 {
88 funset.push_back(llvmModuleSet()->getFunObjVar(item));
89 }
90 pag->callGraph = callGraphBuilder.buildSVFIRCallGraph(funset);
91
92 CHGraph* chg = new CHGraph();
93 CHGBuilder chgbuilder(chg);
94 chgbuilder.buildCHG();
95 pag->setCHG(chg);
96
98 for (Module& M : llvmModuleSet()->getLLVMModules())
99 {
100 for (Module::const_iterator F = M.begin(), E = M.end(); F != E; ++F)
101 {
102 const Function& fun = *F;
103 const FunObjVar* svffun = llvmModuleSet()->getFunObjVar(&fun);
105 if(!fun.isDeclaration())
106 {
112 if (fun.doesNotReturn() == false &&
113 fun.getReturnType()->isVoidTy() == false)
114 {
115 pag->addFunRet(svffun,
116 pag->getValVar(pag->getReturnNode(svffun)));
117 }
118
121 for (Function::const_arg_iterator I = fun.arg_begin(), E = fun.arg_end();
122 I != E; ++I)
123 {
124 setCurrentLocation(&*I,&fun.getEntryBlock());
125 NodeID argValNodeId = llvmModuleSet()->getValueNode(&*I);
126 // if this is the function does not have caller (e.g. main)
127 // or a dead function, shall we create a black hole address edge for it?
128 // it is (1) too conservative, and (2) make FormalParmVFGNode defined at blackhole address PAGEdge.
129 // if(SVFUtil::ArgInNoCallerFunction(&*I)) {
130 // if(I->getType()->isPointerTy())
131 // addBlackHoleAddrEdge(argValNodeId);
132 //}
133 pag->addFunArgs(svffun,pag->getValVar(argValNodeId));
134 }
135 auto returns = aggregateReturnParams.find(&fun);
136 if (returns != aggregateReturnParams.end())
137 for (const auto& field : returns->second)
138 pag->addFunArgs(svffun, pag->getValVar(field.second));
139 }
140 for (Function::const_iterator bit = fun.begin(), ebit = fun.end();
141 bit != ebit; ++bit)
142 {
143 const BasicBlock& bb = *bit;
144 for (BasicBlock::const_iterator it = bb.begin(), eit = bb.end();
145 it != eit; ++it)
146 {
147 const Instruction& inst = *it;
148 setCurrentLocation(&inst,&bb);
149 visit(const_cast<Instruction&>(inst));
150 }
151 }
152 }
153 }
154
155 sanityCheck();
156
157 pag->initialiseCandidatePointers();
158
159 pag->setNodeNumAfterPAGBuild(pag->getTotalNodeNum());
160
161 // dump SVFIR
162 if (Options::PAGDotGraph())
163 pag->dump("svfir_initial");
164
165 // print to command line of the SVFIR graph
166 if (Options::PAGPrint())
167 pag->print();
168
169 // dump ICFG
170 if (Options::DumpICFG())
171 pag->getICFG()->dump("icfg_initial");
172
173 if (Options::LoopAnalysis())
174 {
175 LLVMLoopAnalysis loopAnalysis;
176 loopAnalysis.build(pag->getICFG());
177 }
178
179 // dump SVFIR as JSON
180 if (!Options::DumpJson().empty())
181 {
182 assert(false && "please implement SVFIRWriter::writeJsonToPath");
183 }
184
185 double endTime = SVFStat::getClk(true);
186 SVFStat::timeOfBuildingSVFIR = (endTime - startTime) / TIMEINTERVAL;
187
188 return pag;
189}

219void SVFIRBuilder::initSVFBasicBlock(const Function* func)
220{
221 FunObjVar *svfFun = const_cast<FunObjVar *>(llvmModuleSet()->getFunObjVar(func));
222 for (Function::const_iterator bit = func->begin(), ebit = func->end(); bit != ebit; ++bit)
223 {
224 const BasicBlock* bb = &*bit;
225 SVFBasicBlock* svfbb = llvmModuleSet()->getSVFBasicBlock(bb);
226 for (succ_const_iterator succ_it = succ_begin(bb); succ_it != succ_end(bb); succ_it++)
227 {
228 const SVFBasicBlock* svf_scc_bb = llvmModuleSet()->getSVFBasicBlock(*succ_it);
229 svfbb->addSuccBasicBlock(svf_scc_bb);
230 }
231 for (const_pred_iterator pred_it = pred_begin(bb); pred_it != pred_end(bb); pred_it++)
232 {
233 const SVFBasicBlock* svf_pred_bb = llvmModuleSet()->getSVFBasicBlock(*pred_it);
234 svfbb->addPredBasicBlock(svf_pred_bb);
235 }
236
238 if (svfbb->getSuccessors().empty())
239 {
240 if (LLVMUtil::basicBlockHasRetInst(bb))
241 {
242 assert((LLVMUtil::functionDoesNotRet(func) ||
243 SVFUtil::isa<ReturnInst>(bb->back())) &&
244 "last inst must be return inst");
245 svfFun->setExitBlock(svfbb);
246 }
247 }
248 }
249 // For no return functions, we set the last block as exit BB
250 // This ensures that each function that has definition must have an exit BB
251 if (svfFun->hasBasicBlock() && svfFun->exitBlock == nullptr)
252 {
253 SVFBasicBlock* retBB = const_cast<SVFBasicBlock*>(svfFun->back());
254 assert((LLVMUtil::functionDoesNotRet(func) ||
255 SVFUtil::isa<ReturnInst>(&func->back().back())) &&
256 "last inst must be return inst");
257 svfFun->setExitBlock(retBB);
258 }
259}

262void SVFIRBuilder::initDomTree(FunObjVar* svffun, const Function* fun)
263{
264 if (fun->isDeclaration())
265 return;
266 //process and stored dt & df
267 DominanceFrontier df;
268 DominatorTree& dt = llvmModuleSet()->getDomTree(fun);
269 df.analyze(dt);
270 LoopInfo loopInfo = LoopInfo(dt);
271 PostDominatorTree pdt = PostDominatorTree(const_cast<Function&>(*fun));
272 SVFLoopAndDomInfo* ld = svffun->getLoopAndDomInfo();
273
274 Map<const SVFBasicBlock*,Set<const SVFBasicBlock*>> & dfBBsMap = ld->getDomFrontierMap();
275 for (DominanceFrontierBase::const_iterator dfIter = df.begin(), eDfIter = df.end(); dfIter != eDfIter; dfIter++)
276 {
277 const BasicBlock* keyBB = dfIter->first;
278#if LLVM_VERSION_MAJOR > 16
279 const llvm::SetVector<llvm::BasicBlock* >& domSet = dfIter->second;
280#else
281 const std::set<BasicBlock* >& domSet = dfIter->second;
282#endif
283 Set<const SVFBasicBlock*>& valueBasicBlocks = dfBBsMap[llvmModuleSet()->getSVFBasicBlock(keyBB)];
284 for (const BasicBlock* bbValue:domSet)
285 {
286 valueBasicBlocks.insert(llvmModuleSet()->getSVFBasicBlock(bbValue));
287 }
288 }
289 std::vector<const SVFBasicBlock*> reachableBBs;
290 LLVMUtil::getFunReachableBBs(fun, reachableBBs);
291 ld->setReachableBBs(reachableBBs);
292
293 for (Function::const_iterator bit = fun->begin(), beit = fun->end(); bit!=beit; ++bit)
294 {
295 const BasicBlock &bb = *bit;
296 SVFBasicBlock* svfBB = llvmModuleSet()->getSVFBasicBlock(&bb);
297 if (DomTreeNode* dtNode = dt.getNode(&bb))
298 {
299 SVFLoopAndDomInfo::BBSet& bbSet = ld->getDomTreeMap()[svfBB];
300 for (const auto domBB : *dtNode)
301 {
302 const auto* domSVFBB = llvmModuleSet()->getSVFBasicBlock(domBB->getBlock());
303 bbSet.insert(domSVFBB);
304 }
305 }
306
307 if (DomTreeNode* pdtNode = pdt.getNode(&bb))
308 {
309 u32_t level = pdtNode->getLevel();
310 ld->getBBPDomLevel()[svfBB] = level;
311 BasicBlock* idomBB = pdtNode->getIDom()->getBlock();
312 const SVFBasicBlock* idom = idomBB == NULL ? NULL: llvmModuleSet()->getSVFBasicBlock(idomBB);
313 ld->getBB2PIdom()[svfBB] = idom;
314
315 SVFLoopAndDomInfo::BBSet& bbSet = ld->getPostDomTreeMap()[svfBB];
316 for (const auto domBB : *pdtNode)
317 {
318 const auto* domSVFBB = llvmModuleSet()->getSVFBasicBlock(domBB->getBlock());
319 bbSet.insert(domSVFBB);
320 }
321 }
322
323 if (const Loop* loop = loopInfo.getLoopFor(&bb))
324 {
325 for (const BasicBlock* loopBlock : loop->getBlocks())
326 {
327 const SVFBasicBlock* loopbb = llvmModuleSet()->getSVFBasicBlock(loopBlock);
328 ld->addToBB2LoopMap(svfBB, loopbb);
329 }
330 }
331 }

334void SVFIRBuilder::createFunObjVars()
335{
336 std::vector<FunObjVar*> funset;
337 // Iterate over all object symbols in the symbol table
338 for (const auto* fun: llvmModuleSet()->getFunctionSet())
339 {
340 u32_t id = llvmModuleSet()->objSyms()[fun];
341 // Debug output for adding object node
342 DBOUT(DPAGBuild, outs() << "add obj node " << id << "\n");
343
344 // Check if the value is a function and add a function object node
345 pag->addFunObjNode(id, pag->getObjTypeInfo(id), nullptr);
346 llvmModuleSet()->LLVMFun2FunObjVar[fun] = cast<FunObjVar>(pag->getGNode(id));
347
348 FunObjVar *funObjVar = SVFUtil::cast<FunObjVar>(pag->getGNode(id));
349 funset.push_back(funObjVar);
350
351 funObjVar->initFunObjVar(fun->isDeclaration(), LLVMUtil::isIntrinsicFun(fun), fun->hasAddressTaken(),
352 LLVMUtil::isUncalledFunction(fun), LLVMUtil::functionDoesNotRet(fun), fun->isVarArg(),
353 SVFUtil::cast<SVFFunctionType>(llvmModuleSet()->getSVFType(fun->getFunctionType())),
354 new SVFLoopAndDomInfo, nullptr, nullptr,
355 {}, nullptr);
356 BasicBlockGraph* bbGraph = new BasicBlockGraph();
357 funObjVar->setBasicBlockGraph(bbGraph);
358
359
360 for (const BasicBlock& bb : *fun)
361 {
362 llvmModuleSet()->addBasicBlock(funObjVar, &bb);
363 }
364
366 for (auto& bb: *funObjVar->bbGraph)
367 {
368 bb.second->setFun(funObjVar);
369 }
370 llvmModuleSet()->addToSVFVar2LLVMValueMap(fun, pag->getGNode(id));
371 }
372
373 initFunObjVar();
374}

376void SVFIRBuilder::initialiseBaseObjVars()
377{
378 // Iterate over all object symbols in the symbol table
379 for (LLVMModuleSet::ValueToIDMapTy::iterator iter =
380 llvmModuleSet()->objSyms().begin(); iter != llvmModuleSet()->objSyms().end();
381 ++iter)
382 {
383 // Debug output for adding object node
384 DBOUT(DPAGBuild, outs() << "add obj node " << iter->second << "\n");
385
386 // Skip blackhole and constant symbols
387 if(iter->second == pag->blackholeSymID() || iter->second == pag->constantSymID())
388 continue;
389
390 // Get the LLVM value corresponding to the symbol
391 const Value* llvmValue = iter->first;
392
393 const ICFGNode* icfgNode = nullptr;
394 if (const Instruction* inst = SVFUtil::dyn_cast<Instruction>(llvmValue))
395 {
396 if(llvmModuleSet()->hasICFGNode(inst))
397 icfgNode = llvmModuleSet()->getICFGNode(inst);
398 }
399
400 // Check if the value is a function and add a function object node
401 if (SVFUtil::dyn_cast<Function>(llvmValue))
402 {
403 // already one
404 }
405 // Check if the value is a heap object and add a heap object node
406 else if (LLVMUtil::isHeapObj(llvmValue))
407 {
408 NodeID id = llvmModuleSet()->getObjectNode(iter->first);
409 pag->addHeapObjNode(iter->second, pag->getObjTypeInfo(id), icfgNode);
410 }
411 // Check if the value is an alloca instruction and add a stack object node
412 else if (LLVMUtil::isStackObj(llvmValue))
413 {
414 NodeID id = llvmModuleSet()->getObjectNode(iter->first);
415 pag->addStackObjNode(iter->second, pag->getObjTypeInfo(id), icfgNode);
416 }
417 else if (auto fpValue = SVFUtil::dyn_cast<ConstantFP>(llvmValue))
418 {
419 NodeID id = llvmModuleSet()->getObjectNode(iter->first);
420 pag->addConstantFPObjNode(iter->second, pag->getObjTypeInfo(id), LLVMUtil::getDoubleValue(fpValue), icfgNode);
421 }
422 else if (auto intValue = SVFUtil::dyn_cast<ConstantInt>(llvmValue))
423 {
424 NodeID id = llvmModuleSet()->getObjectNode(iter->first);
425 pag->addConstantIntObjNode(iter->second, pag->getObjTypeInfo(id), LLVMUtil::getIntegerValue(intValue), icfgNode);
426 }
427 else if (SVFUtil::isa<ConstantPointerNull>(llvmValue))
428 {
429 NodeID id = llvmModuleSet()->getObjectNode(iter->first);
430 pag->addConstantNullPtrObjNode(iter->second, pag->getObjTypeInfo(id), icfgNode);
431 }
432 else if (SVFUtil::isa<GlobalValue>(llvmValue))
433 {
434 NodeID id = llvmModuleSet()->getObjectNode(iter->first);
435 pag->addGlobalObjNode(iter->second, pag->getObjTypeInfo(id), icfgNode);
436 }
437 else if (SVFUtil::isa<ConstantData, ConstantExpr, MetadataAsValue, BlockAddress, ConstantAggregate>(llvmValue))
438 {
439 NodeID id = llvmModuleSet()->getObjectNode(iter->first);
440 pag->addConstantDataObjNode(iter->second, pag->getObjTypeInfo(id), icfgNode);
441 }
442 // Add a generic object node for other types of values
443 else
444 {
445 NodeID id = llvmModuleSet()->getObjectNode(iter->first);
446 pag->addObjNode(iter->second, pag->getObjTypeInfo(id), icfgNode);
447 }
448 llvmModuleSet()->addToSVFVar2LLVMValueMap(llvmValue, pag->getGNode(iter->second));
449 }

453void SVFIRBuilder::initialiseValVars()
454{
455 // Iterate over all value symbols in the symbol table
456 for (LLVMModuleSet::ValueToIDMapTy::iterator iter =
457 llvmModuleSet()->valSyms().begin(); iter != llvmModuleSet()->valSyms().end();
458 ++iter)
459 {
460 // Debug output for adding value node
461 DBOUT(DPAGBuild, outs() << "add val node " << iter->second << "\n");
462
463 // Skip blackhole and null pointer symbols
464 if(iter->second == pag->blkPtrSymID() || iter->second == pag->nullPtrSymID())
465 continue;
466
467 const ICFGNode* icfgNode = nullptr;
468 auto llvmValue = iter->first;
469
470 // Check if the value is a function and get its call graph node
471 if (const Function* func = SVFUtil::dyn_cast<Function>(llvmValue))
472 {
473 pag->addFunValNode(iter->second, icfgNode, llvmModuleSet()->getFunObjVar(func), llvmModuleSet()->getSVFType(llvmValue->getType()));
474 }
475 else if (auto argval = SVFUtil::dyn_cast<Argument>(llvmValue))
476 {
477 // Formal params are defined at FunEntryICFGNode (where CallPE copies actual args).
478 // External (declaration-only) functions have no entry node, so keep nullptr.
479 const FunObjVar* funObj = llvmModuleSet()->getFunObjVar(argval->getParent());
480 const ICFGNode* entryNode = funObj->isDeclaration() ? nullptr : pag->getICFG()->getFunEntryICFGNode(funObj);
481 pag->addArgValNode(
482 iter->second, argval->getArgNo(), entryNode,
483 funObj, llvmModuleSet()->getSVFType(llvmValue->getType()));
484 if (!argval->hasName())
485 pag->getGNode(iter->second)->setName("arg_" + std::to_string(argval->getArgNo()));
486 }
487 else if (auto fpValue = SVFUtil::dyn_cast<ConstantFP>(llvmValue))
488 {
489 pag->addConstantFPValNode(iter->second, LLVMUtil::getDoubleValue(fpValue), icfgNode, llvmModuleSet()->getSVFType(llvmValue->getType()));
490 }
491 else if (auto intValue = SVFUtil::dyn_cast<ConstantInt>(llvmValue))
492 {
493 pag->addConstantIntValNode(iter->second, LLVMUtil::getIntegerValue(intValue), icfgNode, llvmModuleSet()->getSVFType(llvmValue->getType()));
494 }
495 else if (SVFUtil::isa<ConstantPointerNull>(llvmValue))
496 {
497 pag->addConstantNullPtrValNode(iter->second, icfgNode, llvmModuleSet()->getSVFType(llvmValue->getType()));
498 }
499 else if (SVFUtil::isa<GlobalValue>(llvmValue))
500 {
501 // Global variables are defined at the global ICFG node.
502 pag->addGlobalValNode(iter->second, pag->getICFG()->getGlobalICFGNode(),
503 llvmModuleSet()->getSVFType(llvmValue->getType()));
504 }
505 else if (SVFUtil::isa<ConstantData, ConstantExpr, MetadataAsValue, BlockAddress, ConstantAggregate>(llvmValue))
506 {
507 pag->addConstantDataValNode(iter->second, icfgNode, llvmModuleSet()->getSVFType(llvmValue->getType()));
508 }
509 else if (SVFUtil::isa<InlineAsm>(llvmValue) ||
510 SVFUtil::isa<DSOLocalEquivalent>(llvmValue) ||
511 SVFUtil::isa<NoCFIValue>(llvmValue))
512 {
513 pag->addAsmPCValNode(iter->second, llvmModuleSet()->getSVFType(llvmValue->getType()));
514 }
515 else if (const Instruction* inst = SVFUtil::dyn_cast<Instruction>(llvmValue))
516 {
517 if (LLVMUtil::isIntrinsicInst(inst))
518 pag->addIntrinsicValNode(iter->second, llvmModuleSet()->getSVFType(llvmValue->getType()));
519 else
520 {
521 assert(llvmModuleSet()->hasICFGNode(inst) && "LLVM instruction is not associated with an ICFGNode");
522 icfgNode = llvmModuleSet()->getICFGNode(inst);
523 pag->addValNode(iter->second, llvmModuleSet()->getSVFType(llvmValue->getType()), icfgNode);
524 }
525 }
526 llvmModuleSet()->addToSVFVar2LLVMValueMap(llvmValue,
527 pag->getGNode(iter->second));
528 }

535void SVFIRBuilder::initialiseNodes()
536{
537 DBOUT(DPAGBuild, outs() << "Initialise SVFIR Nodes ...\n");
538
539
540 pag->addBlackholeObjNode();
541 pag->addConstantObjNode();
542 pag->addBlackholePtrNode();
543 addNullPtrNode();
544
545 initialiseBaseObjVars();
546 initialiseValVars();
547
548 for (LLVMModuleSet::FunToIDMapTy::iterator iter =
549 llvmModuleSet()->retSyms().begin(); iter != llvmModuleSet()->retSyms().end();
550 ++iter)
551 {
552 const Value* llvmValue = iter->first;
553 // retSyms keys are Function*, not Instruction, so dyn_cast<Instruction> always fails.
554 // RetValPN represents the callee's return value, defined at FunExitICFGNode.
555 // External functions have no exit node, so keep nullptr.
556 const FunObjVar* funObjVar = llvmModuleSet()->getFunObjVar(SVFUtil::cast<Function>(llvmValue));
557 const ICFGNode* icfgNode = funObjVar->isDeclaration() ? nullptr : pag->getICFG()->getFunExitICFGNode(funObjVar);
558 DBOUT(DPAGBuild, outs() << "add ret node " << iter->second << "\n");
559 pag->addRetNode(iter->second,
560 funObjVar,
561 llvmModuleSet()->getSVFType(iter->first->getType()), icfgNode);
562 llvmModuleSet()->addToSVFVar2LLVMValueMap(llvmValue, pag->getGNode(iter->second));
563 pag->returnFunObjSymMap[funObjVar] = iter->second;
564 }
565
566 for (LLVMModuleSet::FunToIDMapTy::iterator iter =
567 llvmModuleSet()->varargSyms().begin();
568 iter != llvmModuleSet()->varargSyms().end(); ++iter)
569 {
570 const Value* llvmValue = iter->first;
571 // varargSyms keys are Function*, not Instruction.
572 // Variadic arguments are received at the function entry point.
573 // External functions have no entry node, so keep nullptr.
574 const FunObjVar* funObjVar = llvmModuleSet()->getFunObjVar(SVFUtil::cast<Function>(llvmValue));
575 const ICFGNode* icfgNode = funObjVar->isDeclaration() ? nullptr : pag->getICFG()->getFunEntryICFGNode(funObjVar);
576 DBOUT(DPAGBuild, outs() << "add vararg node " << iter->second << "\n");
577 pag->addVarargNode(iter->second,
578 funObjVar,
579 llvmModuleSet()->getSVFType(iter->first->getType()), icfgNode);
580 llvmModuleSet()->addToSVFVar2LLVMValueMap(llvmValue, pag->getGNode(iter->second));
581 pag->varargFunObjSymMap[funObjVar] = iter->second;
582 }
583
585 for (LLVMModuleSet::ValueToIDMapTy::iterator iter =
586 llvmModuleSet()->objSyms().begin(); iter != llvmModuleSet()->objSyms().end(); ++iter)
587 {
588 DBOUT(DPAGBuild, outs() << "add address edges for constant node " << iter->second << "\n");
589 const Value* val = iter->first;
590 if (isConstantObjSym(val))
591 {
592 NodeID ptr = llvmModuleSet()->getValueNode(val);
593 if(ptr!= pag->getBlkPtr() && ptr!= pag->getNullPtr())
594 {
595 setCurrentLocation(val, (SVFBasicBlock*) nullptr);
596 addAddrEdge(iter->second, ptr);
597 }
598 }
599 }
600
601 assert(pag->getTotalNodeNum() >= pag->getTotalSymNum()
602 && "not all node have been initialized!!!");
603
605 for (auto& fun: llvmModuleSet()->getFunctionSet())
606 {
607 for (const Argument& arg : fun->args())
608 {
609 const_cast<FunObjVar*>(llvmModuleSet()->getFunObjVar(fun))->addArgument(SVFUtil::cast<ArgValVar>(
610 pag->getGNode(llvmModuleSet()->getValueNode(&arg))));
611 }
612 }

636bool SVFIRBuilder::computeGepOffset(const User *V, AccessPath& ap)
637{
638 assert(V);
639
640 const llvm::GEPOperator *gepOp = SVFUtil::dyn_cast<const llvm::GEPOperator>(V);
641 DataLayout * dataLayout = getDataLayout(llvmModuleSet()->getMainLLVMModule());
642 llvm::APInt byteOffset(dataLayout->getIndexSizeInBits(gepOp->getPointerAddressSpace()),0,true);
643 if(gepOp && dataLayout && gepOp->accumulateConstantOffset(*dataLayout,byteOffset))
644 {
645 //s32_t bo = byteOffset.getSExtValue();
646 }
647
648 bool isConst = true;
649
650 bool prevPtrOperand = false;
651 for (bridge_gep_iterator gi = bridge_gep_begin(*V), ge = bridge_gep_end(*V);
652 gi != ge; ++gi)
653 {
654 const Type* gepTy = *gi;
655 const SVFType* svfGepTy = llvmModuleSet()->getSVFType(gepTy);
656
657 assert((prevPtrOperand && svfGepTy->isPointerTy()) == false &&
658 "Expect no more than one gep operand to be of a pointer type");
659 if(!prevPtrOperand && svfGepTy->isPointerTy()) prevPtrOperand = true;
660 const Value* offsetVal = gi.getOperand();
661 assert(gepTy != offsetVal->getType() && "iteration and operand have the same type?");
662
663 const ArrayType* inferredPtrArrayTy = nullptr;
664 const SVFType* idxGepTy = svfGepTy;
665 if (svfGepTy->isPointerTy() && gepOp->getSourceElementType()->isSingleValueType())
666 {
667 const Type* baseObjType =
668 LLVMModuleSet::getLLVMModuleSet()->getTypeInference()->inferObjType(gepOp->getPointerOperand());
669 if (const auto* arrTy = SVFUtil::dyn_cast<ArrayType>(baseObjType))
670 {
671 if (arrTy->getElementType()->isPointerTy())
672 {
673 inferredPtrArrayTy = arrTy;
674 idxGepTy = llvmModuleSet()->getSVFType(arrTy);
675 }
676 }
677 }
678
679 ap.addOffsetVarAndGepTypePair(getPAG()->getValVar(llvmModuleSet()->getValueNode(offsetVal)), idxGepTy);
680
681 //The int value of the current index operand
682 const ConstantInt* op = SVFUtil::dyn_cast<ConstantInt>(offsetVal);
683
684 // if Options::ModelConsts() is disabled. We will treat whole array as one,
685 // but we can distinguish different field of an array of struct, e.g. s[1].f1 is different from s[0].f2
686 if(const ArrayType* arrTy = SVFUtil::dyn_cast<ArrayType>(gepTy))
687 {
688 if (!Options::ModelArrays() && arrTy->getElementType()->isPointerTy())
689 continue;
690 if(!op || (arrTy->getArrayNumElements() <= (u32_t)LLVMUtil::getIntegerValue(op).first))
691 continue;
692 APOffset idx = (u32_t)LLVMUtil::getIntegerValue(op).first;
693 u32_t offset = pag->getFlattenedElemIdx(llvmModuleSet()->getSVFType(arrTy), idx);
694 ap.setFldIdx(ap.getConstantStructFldIdx() + offset);
695 }
696 else if (const StructType *ST = SVFUtil::dyn_cast<StructType>(gepTy))
697 {
698 assert(op && "non-const offset accessing a struct");
699 // guard against negative or out-of-bounds struct indices
700 // (e.g. rust hashbrown bucket back-offset: gep { ... }, ptr %p, i64 -1)
701 // a negative i64 wraps to a huge uint64_t that overflows u32_t,
702 // creating an invalid field index that severs points-to tracking
703 uint64_t rawIdx = LLVMUtil::getIntegerValue(op).first;
704 if (rawIdx >= ST->getNumElements())
705 {
706 isConst = false;
707 continue;
708 }
709 APOffset idx = (u32_t)rawIdx;
710 u32_t offset = pag->getFlattenedElemIdx(llvmModuleSet()->getSVFType(ST), idx);
711 ap.setFldIdx(ap.getConstantStructFldIdx() + offset);
712 }
713 else if (gepTy->isSingleValueType())
714 {
715 if (inferredPtrArrayTy)
716 {
717 if (!op || (inferredPtrArrayTy->getArrayNumElements() <= (u32_t)LLVMUtil::getIntegerValue(op).first))
718 continue;
719 APOffset idx = (u32_t)LLVMUtil::getIntegerValue(op).first;
720 u32_t offset = pag->getFlattenedElemIdx(llvmModuleSet()->getSVFType(inferredPtrArrayTy), idx);
721 ap.setFldIdx(ap.getConstantStructFldIdx() + offset);
722 continue;
723 }
724 // If it's a non-constant offset access
725 // If its point-to target is struct or array, it's likely an array accessing (%result = gep %struct.A* %a, i32 %non-const-index)
726 // If its point-to target is single value (pointer arithmetic), then it's a variant gep (%result = gep i8* %p, i32 %non-const-index)
727 if(!op && gepTy->isPointerTy() && gepOp->getSourceElementType()->isSingleValueType())
728 {
729 isConst = false;
730 }
731
732 // The actual index
733 //s32_t idx = op->getSExtValue();
734
735 // For pointer arithmetic we ignore the byte offset
736 // consider using inferFieldIdxFromByteOffset(geopOp,dataLayout,ap,idx)?
737 // ap.setFldIdx(ap.getConstantFieldIdx() + inferFieldIdxFromByteOffset(geopOp,idx));
738 }
739 }
740 return isConst;
741}

746void SVFIRBuilder::processCE(const Value* val)
747{
748 if (const Constant* ref = SVFUtil::dyn_cast<Constant>(val))
749 {
750 if (const ConstantExpr* gepce = isGepConstantExpr(ref))
751 {
752 DBOUT(DPAGBuild, outs() << "handle gep constant expression "
753 << LLVMUtil::dumpValue(ref) << "\n");
754 const Constant* opnd = gepce->getOperand(0);
755 // handle recursive constant express case (gep (bitcast (gep X 1)) 1)
756 processCE(opnd);
757 auto &GEPOp = llvm::cast<llvm::GEPOperator>(*gepce);
758 Type *pType = GEPOp.getSourceElementType();
759 AccessPath ap(0, llvmModuleSet()->getSVFType(pType));
760 bool constGep = computeGepOffset(gepce, ap);
761 // must invoke pag methods here, otherwise it will be a dead recursion cycle
762 const Value* cval = getCurrentValue();
763 const SVFBasicBlock* cbb = getCurrentBB();
764 setCurrentLocation(gepce, (SVFBasicBlock*) nullptr);
765 /*
766 * The gep edge created are like constexpr (same edge may appear at multiple callsites)
767 * so bb/inst of this edge may be rewritten several times, we treat it as global here.
768 */
769 addGepEdge(llvmModuleSet()->getValueNode(opnd), llvmModuleSet()->getValueNode(gepce), ap, constGep);
770 setCurrentLocation(cval, cbb);
771 }
772 else if (const ConstantExpr* castce = isCastConstantExpr(ref))
773 {
774 DBOUT(DPAGBuild, outs() << "handle cast constant expression "
775 << LLVMUtil::dumpValue(ref) << "\n");
776 const Constant* opnd = castce->getOperand(0);
777 processCE(opnd);
778 const Value* cval = getCurrentValue();
779 const SVFBasicBlock* cbb = getCurrentBB();
780 setCurrentLocation(castce, (SVFBasicBlock*) nullptr);
781 addCopyEdge(llvmModuleSet()->getValueNode(opnd), llvmModuleSet()->getValueNode(castce), CopyStmt::BITCAST);
782 setCurrentLocation(cval, cbb);
783 }
784 else if (const ConstantExpr* selectce = isSelectConstantExpr(ref))
785 {
786 DBOUT(DPAGBuild, outs() << "handle select constant expression "
787 << LLVMUtil::dumpValue(ref) << "\n");
788 const Constant* src1 = selectce->getOperand(1);
789 const Constant* src2 = selectce->getOperand(2);
790 processCE(src1);
791 processCE(src2);
792 const Value* cval = getCurrentValue();
793 const SVFBasicBlock* cbb = getCurrentBB();
794 setCurrentLocation(selectce, (SVFBasicBlock*) nullptr);
795 NodeID cond = llvmModuleSet()->getValueNode(selectce->getOperand(0));
796 NodeID nsrc1 = llvmModuleSet()->getValueNode(src1);
797 NodeID nsrc2 = llvmModuleSet()->getValueNode(src2);
798 NodeID nres = llvmModuleSet()->getValueNode(selectce);
799 addSelectStmt(nres,nsrc1, nsrc2, cond);
800 setCurrentLocation(cval, cbb);
801 }
802 // if we meet a int2ptr, then it points-to black hole
803 else if (const ConstantExpr* int2Ptrce = isInt2PtrConstantExpr(ref))
804 {
805 const Constant* opnd = int2Ptrce->getOperand(0);
806 processCE(opnd);
807 const SVFBasicBlock* cbb = getCurrentBB();
808 const Value* cval = getCurrentValue();
809 setCurrentLocation(int2Ptrce, (SVFBasicBlock*) nullptr);
810 addCopyEdge(llvmModuleSet()->getValueNode(opnd), llvmModuleSet()->getValueNode(int2Ptrce), CopyStmt::INTTOPTR);
811 setCurrentLocation(cval, cbb);
812 }
813 else if (const ConstantExpr* ptr2Intce = isPtr2IntConstantExpr(ref))
814 {
815 const Constant* opnd = ptr2Intce->getOperand(0);
816 processCE(opnd);
817 const SVFBasicBlock* cbb = getCurrentBB();
818 const Value* cval = getCurrentValue();
819 setCurrentLocation(ptr2Intce, (SVFBasicBlock*) nullptr);
820 addCopyEdge(llvmModuleSet()->getValueNode(opnd), llvmModuleSet()->getValueNode(ptr2Intce), CopyStmt::PTRTOINT);
821 setCurrentLocation(cval, cbb);
822 }
823 else if(isTruncConstantExpr(ref) || isCmpConstantExpr(ref))
824 {
825 // we don't handle trunc and cmp instruction for now
826 const Value* cval = getCurrentValue();
827 const SVFBasicBlock* cbb = getCurrentBB();
828 setCurrentLocation(ref, (SVFBasicBlock*) nullptr);
829 NodeID dst = llvmModuleSet()->getValueNode(ref);
830 addBlackHoleAddrEdge(dst);
831 setCurrentLocation(cval, cbb);
832 }
833 else if (isBinaryConstantExpr(ref))
834 {
835 // we don't handle binary constant expression like add(x,y) now
836 const Value* cval = getCurrentValue();
837 const SVFBasicBlock* cbb = getCurrentBB();
838 setCurrentLocation(ref, (SVFBasicBlock*) nullptr);
839 NodeID dst = llvmModuleSet()->getValueNode(ref);
840 addBlackHoleAddrEdge(dst);
841 setCurrentLocation(cval, cbb);
842 }
843 else if (isUnaryConstantExpr(ref))
844 {
845 // we don't handle unary constant expression like fneg(x) now
846 const Value* cval = getCurrentValue();
847 const SVFBasicBlock* cbb = getCurrentBB();
848 setCurrentLocation(ref, (SVFBasicBlock*) nullptr);
849 NodeID dst = llvmModuleSet()->getValueNode(ref);
850 addBlackHoleAddrEdge(dst);
851 setCurrentLocation(cval, cbb);
852 }
853 else if (SVFUtil::isa<ConstantAggregate>(ref))
854 {
855 // we don't handle constant aggregate like constant vectors
856 }
857 else if (SVFUtil::isa<BlockAddress>(ref))
858 {
859 // blockaddress instruction (e.g. i8* blockaddress(@run_vm, %182))
860 // is treated as constant data object for now, see LLVMUtil.h:397, SymbolTableInfo.cpp:674 and SVFIRBuilder.cpp:183-194
861 const Value* cval = getCurrentValue();
862 const SVFBasicBlock* cbb = getCurrentBB();
863 setCurrentLocation(ref, (SVFBasicBlock*) nullptr);
864 NodeID dst = llvmModuleSet()->getValueNode(ref);
865 addAddrEdge(pag->getConstantNode(), dst);
866 setCurrentLocation(cval, cbb);
867 }
868 else
869 {
870 if(SVFUtil::isa<ConstantExpr>(val))
871 assert(false && "we don't handle all other constant expression for now!");
872 }
873 }

896/*For global variable initialization
897 * Give a simple global variable
898 * int x = 10; // store 10 x (constant, non pointer) |
899 * int *y = &x; // store x y (pointer type)
900 * Given a struct
901 * struct Z { int s; int *t;};
902 * Global initialization:
903 * struct Z z = {10,&x}; // store x z.t (struct type)
904 * struct Z *m = &z; // store z m (pointer type)
905 * struct Z n = {10,&z.s}; // store z.s n , &z.s constant expression (constant expression)
906 */
907void SVFIRBuilder::InitialGlobal(const GlobalVariable *gvar, Constant *C,
908 u32_t offset)
909{
910 DBOUT(DPAGBuild, outs() << "global " << LLVMUtil::dumpValue(gvar)
911 << " constant initializer: "
912 << LLVMUtil::dumpValue(C) << "\n");
913 if (C->getType()->isSingleValueType())
914 {
915 NodeID src = getValueNode(C);
916 // get the field value if it is available, otherwise we create a dummy field node.
917 setCurrentLocation(gvar, (SVFBasicBlock*) nullptr);
918 NodeID field = getGlobalVarField(gvar, offset, llvmModuleSet()->getSVFType(C->getType()));
919
920 if (SVFUtil::isa<GlobalVariable, Function>(C))
921 {
922 setCurrentLocation(C, (SVFBasicBlock*) nullptr);
923 addStoreEdge(src, field);
924 }
925 else if (SVFUtil::isa<ConstantExpr>(C))
926 {
927 // add gep edge of C1 itself is a constant expression
928 processCE(C);
929 setCurrentLocation(C, (SVFBasicBlock*) nullptr);
930 addStoreEdge(src, field);
931 }
932 else if (SVFUtil::isa<BlockAddress>(C))
933 {
934 // blockaddress instruction (e.g. i8* blockaddress(@run_vm, %182))
935 // is treated as constant data object for now, see LLVMUtil.h:397, SymbolTableInfo.cpp:674 and SVFIRBuilder.cpp:183-194
936 processCE(C);
937 setCurrentLocation(C, (SVFBasicBlock*) nullptr);
938 addAddrEdge(pag->getConstantNode(), src);
939 }
940 else
941 {
942 setCurrentLocation(C, (SVFBasicBlock*) nullptr);
943 addStoreEdge(src, field);
945 if (C->getType()->isPtrOrPtrVectorTy() && src != pag->getNullPtr())
946 addCopyEdge(pag->getNullPtr(), src, CopyStmt::COPYVAL);
947 }
948 }
949 else if (SVFUtil::isa<ConstantArray, ConstantStruct>(C))
950 {
951 if(cppUtil::isValVtbl(gvar) && !Options::VtableInSVFIR())
952 return;
953 for (u32_t i = 0, e = C->getNumOperands(); i != e; i++)
954 {
955 u32_t off = pag->getFlattenedElemIdx(llvmModuleSet()->getSVFType(C->getType()), i);
956 InitialGlobal(gvar, SVFUtil::cast<Constant>(C->getOperand(i)), offset + off);
957 }
958 }
959 else if(ConstantData* data = SVFUtil::dyn_cast<ConstantData>(C))
960 {
961 if(Options::ModelConsts())
962 {
963 if(ConstantDataSequential* seq = SVFUtil::dyn_cast<ConstantDataSequential>(data))
964 {
965 for(u32_t i = 0; i < seq->getNumElements(); i++)
966 {
967 u32_t off = pag->getFlattenedElemIdx(llvmModuleSet()->getSVFType(C->getType()), i);
968 Constant* ct = seq->getElementAsConstant(i);
969 InitialGlobal(gvar, ct, offset + off);
970 }
971 }
972 else
973 {
974 assert((SVFUtil::isa<ConstantAggregateZero, UndefValue>(data)) && "Single value type data should have been handled!");
975 }
976 }
977 }
978 else
979 {
980 //TODO:assert(SVFUtil::isa<ConstantVector>(C),"what else do we have");
981 }
982}

988{
989
991 for (Module &M : llvmModuleSet()->getLLVMModules())
992 {
993 for (Module::global_iterator I = M.global_begin(), E = M.global_end(); I != E; ++I)
994 {
995 GlobalVariable *gvar = &*I;
996 NodeID idx = getValueNode(gvar);
997 NodeID obj = getObjectNode(gvar);
998
999 setCurrentLocation(gvar, (SVFBasicBlock*) nullptr);
1000 addAddrEdge(obj, idx);
1001
1002 if (gvar->hasInitializer())
1003 {
1004 Constant *C = gvar->getInitializer();
1005 DBOUT(DPAGBuild, outs() << "add global var node "
1006 << LLVMUtil::dumpValue(gvar) << "\n");
1007 InitialGlobal(gvar, C, 0);
1008 }
1009 }
1010
1011
1013 for (Module::const_iterator I = M.begin(), E = M.end(); I != E; ++I)
1014 {
1015 const Function* fun = &*I;
1016 NodeID idx = getValueNode(fun);
1017 NodeID obj = getObjectNode(fun);
1018
1019 DBOUT(DPAGBuild, outs() << "add global function node " << fun->getName().str() << "\n");
1020 setCurrentLocation(fun, (SVFBasicBlock*) nullptr);
1021 addAddrEdge(obj, idx);
1022 }
1023
1024 // Handle global aliases (due to linkage of multiple bc files), e.g., @x = internal alias @y. We need to add a copy from y to x.
1025 for (Module::alias_iterator I = M.alias_begin(), E = M.alias_end(); I != E; I++)
1026 {

1028 NodeID dst = llvmModuleSet()->getValueNode(alias);
1029 NodeID src = llvmModuleSet()->getValueNode(alias->getAliasee());
1030 processCE(alias->getAliasee());
1031 setCurrentLocation(alias, (SVFBasicBlock*) nullptr);
1032 addCopyEdge(src, dst, CopyStmt::COPYVAL);
1033 }
1034 }

1254/*
1255 * Visit callsites
1256 */
1257void SVFIRBuilder::visitCallSite(CallBase* cs)
1258{
1259
1260 // skip llvm intrinsics
1261 if(isIntrinsicInst(cs))
1262 return;
1263
1264 DBOUT(DPAGBuild,
1265 outs() << "process callsite " << LLVMUtil::dumpValue(cs) << "\n");
1266
1267 CallICFGNode* callBlockNode = llvmModuleSet()->getCallICFGNode(cs);
1268 RetICFGNode* retBlockNode = llvmModuleSet()->getRetICFGNode(cs);
1269
1270 pag->addCallSite(callBlockNode);
1271
1273 for (u32_t i = 0; i < cs->arg_size(); i++)
1274 pag->addCallSiteArgs(
1275 callBlockNode,
1276 pag->getValVar(getValueNode(cs->getArgOperand(i))));
1277
1278 if(!cs->getType()->isVoidTy())
1279 pag->addCallSiteRets(retBlockNode,pag->getValVar(getValueNode(cs)));
1280
1281 if (const auto* call = SVFUtil::dyn_cast<CallInst>(cs))
1282 addAggregateCallFields(call);
1283
1284 if (callBlockNode->isVirtualCall())
1285 {
1286 const Value* value = cppUtil::getVCallVtblPtr(cs);
1287 callBlockNode->setVtablePtr(pag->getGNode(getValueNode(value)));
1288 }
1289 if (const Function *callee = LLVMUtil::getCallee(cs))
1290 {
1291 if (LLVMUtil::isExtCall(callee))
1292 {
1293 handleExtCall(cs, callee);

1337{

1365bool SVFIRBuilder::getAggregateFieldOperands(const AggregateFieldKey& field,
1366 std::vector<AggregateFieldKey>& operands)
1367{
1368 const Value* value = field.first;
1369 const AggregateField& path = field.second;
1370 if (path.empty())
1371 return value->getType()->isPointerTy() && !SVFUtil::isa<ConstantExpr>(value) &&
1372 !SVFUtil::isa<UndefValue>(value) && llvmModuleSet()->hasValueNode(value);
1373 if (const auto* insert = SVFUtil::dyn_cast<InsertValueInst>(value))
1374 {
1375 const auto insertedPath = insert->getIndices();
1376 const size_t commonLength = std::min(insertedPath.size(), path.size());
1377 if (!std::equal(insertedPath.begin(), insertedPath.begin() + commonLength, path.begin()))
1378 operands.emplace_back(insert->getAggregateOperand(), path);
1379 else if (insertedPath.size() <= path.size())
1380 operands.emplace_back(insert->getInsertedValueOperand(),
1381 AggregateField(path.begin() + insertedPath.size(), path.end()));
1382 else
1383 return false;
1384 }
1385 else if (const auto* extract = SVFUtil::dyn_cast<ExtractValueInst>(value))
1386 {
1387 AggregateField outer(extract->idx_begin(), extract->idx_end());
1388 outer.insert(outer.end(), path.begin(), path.end());
1389 operands.emplace_back(extract->getAggregateOperand(), std::move(outer));
1390 }
1391 else if (const auto* phi = SVFUtil::dyn_cast<PHINode>(value))
1392 {
1393 for (const Value* incoming : phi->incoming_values())
1394 operands.emplace_back(incoming, path);
1395 }
1396 else if (const auto* select = SVFUtil::dyn_cast<SelectInst>(value))
1397 {
1398 if (!select->getCondition()->getType()->isIntegerTy(1))
1399 return false;
1400 operands.emplace_back(select->getTrueValue(), path);
1401 operands.emplace_back(select->getFalseValue(), path);
1402 }
1403 else if (const auto* freeze = SVFUtil::dyn_cast<FreezeInst>(value))
1404 operands.emplace_back(freeze->getOperand(0), path);
1405 else if (const auto* insert = SVFUtil::dyn_cast<InsertElementInst>(value))
1406 {
1407 const auto* vector = SVFUtil::dyn_cast<llvm::FixedVectorType>(insert->getType());
1408 const auto* index = SVFUtil::dyn_cast<ConstantInt>(insert->getOperand(2));
1409 if (!vector || path.size() != 1 || !index ||
1410 index->getValue().uge(vector->getNumElements()))
1411 return false;
1412 if (index->getZExtValue() == path.front())
1413 operands.emplace_back(insert->getOperand(1), AggregateField{});
1414 else
1415 operands.emplace_back(insert->getOperand(0), path);
1416 }
1417 else if (const auto* shuffle = SVFUtil::dyn_cast<ShuffleVectorInst>(value))
1418 {
1419 const auto* vector = SVFUtil::dyn_cast<llvm::FixedVectorType>(shuffle->getOperand(0)->getType());
1420 if (!vector || path.size() != 1)
1421 return false;
1422 int index = shuffle->getMaskValue(path.front());
1423 if (index < 0)
1424 return false;
1425 unsigned width = vector->getNumElements();
1426 operands.emplace_back(shuffle->getOperand(static_cast<unsigned>(index) / width),
1427 AggregateField{static_cast<unsigned>(index) % width});
1428 }
1429 else if (const auto* constant = SVFUtil::dyn_cast<Constant>(value))
1430 {
1431 for (unsigned index : path)
1432 {
1433 constant = constant->getAggregateElement(index);
1434 if (!constant)
1435 return false;
1436 }
1437 operands.emplace_back(constant, AggregateField{});
1438 }
1439 else
1440 // Aggregate arguments, calls and loads need their own field interfaces.
1441 return false;
1442 return !operands.empty();
1443}

1456bool SVFIRBuilder::getAggregateFieldNode(const Value* aggregate,
1457 const AggregateField& indices, NodeID& node)
1458{
1459 const AggregateFieldKey root{aggregate, indices};
1460 Map<AggregateFieldKey, std::vector<AggregateFieldKey>> definitions;
1461 std::vector<AggregateFieldKey> worklist{root};
1462 while (!worklist.empty())
1463 {
1464 AggregateFieldKey field = std::move(worklist.back());
1465 worklist.pop_back();
1466 if (aggregateFieldNodes.count(field) || definitions.count(field))
1467 continue;
1468 const Type* type = getAggregateFieldType(field.first->getType(), field.second);
1469 if (!type || !type->isPointerTy())
1470 return false;
1471 auto& operands = definitions[field];
1472 if (!getAggregateFieldOperands(field, operands))
1473 return false;
1474 worklist.insert(worklist.end(), operands.begin(), operands.end());
1475 }
1476
1477 // Validate the entire dependency graph before changing SVFIR. Allocate every
1478 // field before wiring definitions, including loop-carried Phi/Select cycles.
1479 for (const auto& definition : definitions)
1480 {
1481 const AggregateFieldKey& field = definition.first;
1482 NodeID id;
1483 if (field.second.empty())
1484 id = getValueNode(field.first);
1485 else if (SVFUtil::isa<Constant>(field.first))
1486 id = getValueNode(definition.second.front().first);
1487 else
1488 {
1489 const auto* producer = SVFUtil::cast<Instruction>(field.first);
1490 const Type* type = getAggregateFieldType(producer->getType(), field.second);
1491 id = NodeIDAllocator::get()->allocateValueId();
1492 pag->addValNode(id, llvmModuleSet()->getSVFType(type),
1493 llvmModuleSet()->getICFGNode(producer));
1494 llvmModuleSet()->addToSVFVar2LLVMValueMap(producer, pag->getGNode(id));
1495 }
1496 aggregateFieldNodes.emplace(field, id);
1497 }
1498
1499 const Value* savedValue = curVal;
1500 const SVFBasicBlock* savedBB = curBB;
1501 for (const auto& definition : definitions)
1502 {
1503 const AggregateFieldKey& field = definition.first;
1504 if (field.second.empty() || SVFUtil::isa<Constant>(field.first))
1505 continue;
1506 const auto* producer = SVFUtil::cast<Instruction>(field.first);
1507 const auto& operands = definition.second;
1508 NodeID dst = aggregateFieldNodes.at(field);
1509 setCurrentLocation(producer, producer->getParent());
1510 if (const auto* phi = SVFUtil::dyn_cast<PHINode>(producer))
1511 {
1512 for (unsigned i = 0; i < operands.size(); ++i)
1513 addPhiStmt(dst, aggregateFieldNodes.at(operands[i]),
1514 llvmModuleSet()->getICFGNode(phi->getIncomingBlock(i)->getTerminator()));
1515 }
1516 else if (const auto* select = SVFUtil::dyn_cast<SelectInst>(producer))
1517 addSelectStmt(dst, aggregateFieldNodes.at(operands[0]),
1518 aggregateFieldNodes.at(operands[1]), getValueNode(select->getCondition()));
1519 else
1520 addCopyEdge(aggregateFieldNodes.at(operands.front()), dst, CopyStmt::COPYVAL);
1521 }
1522 setCurrentLocation(savedValue, savedBB);
1523 node = aggregateFieldNodes.at(root);
1524 return true;
1525}

1529static bool canModelAggregateReturn(const Function* fun)
1530{
1531 if (!fun || LLVMUtil::isExtCall(fun) || fun->isVarArg() ||
1532 fun->doesNotReturn() || !fun->getReturnType()->isAggregateType() ||

1547void SVFIRBuilder::initialiseAggregateReturns()
1548{
1549 std::vector<AggregateFieldKey> worklist;
1550 for (Module& module : llvmModuleSet()->getLLVMModules())
1551 for (const Function& fun : module)
1552 for (const BasicBlock& bb : fun)
1553 for (const Instruction& inst : bb)
1554 {
1555 if (const auto* extract = SVFUtil::dyn_cast<ExtractValueInst>(&inst))
1556 {
1557 if (extract->getType()->isPointerTy())
1558 worklist.emplace_back(extract->getAggregateOperand(),
1559 AggregateField(extract->idx_begin(), extract->idx_end()));
1560 }
1561 else if (const auto* extract = SVFUtil::dyn_cast<ExtractElementInst>(&inst))
1562 {
1563 const auto* vector = SVFUtil::dyn_cast<llvm::FixedVectorType>(extract->getVectorOperandType());
1564 const auto* index = SVFUtil::dyn_cast<ConstantInt>(extract->getIndexOperand());
1565 if (extract->getType()->isPointerTy() && vector && index &&
1566 index->getValue().ult(vector->getNumElements()))
1567 worklist.emplace_back(extract->getVectorOperand(),
1568 AggregateField{static_cast<unsigned>(index->getZExtValue())});
1569 }
1570 }
1571
1572 // Discover callee demands without materializing local field definitions.
1573 // Calls cross into return operands; other producers use the shared grammar.
1574 Set<AggregateFieldKey> visited;
1575 while (!worklist.empty())
1576 {
1577 AggregateFieldKey field = std::move(worklist.back());
1578 worklist.pop_back();
1579 if (!visited.insert(field).second || field.second.empty())
1580 continue;
1581 if (const auto* call = SVFUtil::dyn_cast<CallInst>(field.first))
1582 {
1583 const Function* callee = call->getCalledFunction();
1584 if (!canModelAggregateReturn(callee) ||
1585 !aggregateReturnParams[callee].emplace(field.second, 0).second)
1586 continue;
1587 for (const BasicBlock& bb : *callee)
1588 if (const auto* ret = SVFUtil::dyn_cast<ReturnInst>(bb.getTerminator()))
1589 if (ret->getReturnValue())
1590 worklist.emplace_back(ret->getReturnValue(), field.second);
1591 }
1592 else
1593 {
1594 std::vector<AggregateFieldKey> operands;
1595 if (getAggregateFieldOperands(field, operands))
1596 worklist.insert(worklist.end(), operands.begin(), operands.end());
1597 }
1598 }
1599
1600 const SVFType* pointerType = llvmModuleSet()->getSVFType(
1601 PointerType::getUnqual(llvmModuleSet()->getContext()));
1602 const Value* savedValue = curVal;
1603 const SVFBasicBlock* savedBB = curBB;
1604 // Serialized points-to sets use NodeIDs. Allocate in function order,
1605 // independent of pointer-keyed hash table iteration.
1606 for (const Function* fun : llvmModuleSet()->getFunctionSet())
1607 {
1608 auto found = aggregateReturnParams.find(fun);
1609 if (found == aggregateReturnParams.end())
1610 continue;
1611 auto& fields = found->second;
1612 FunObjVar* svfFun = const_cast<FunObjVar*>(llvmModuleSet()->getFunObjVar(fun));
1613 const FunEntryICFGNode* entry = pag->getICFG()->getFunEntryICFGNode(svfFun);
1614 for (auto& field : fields)
1615 {
1616 NodeID id = NodeIDAllocator::get()->allocateValueId();
1617 pag->addArgValNode(id, svfFun->arg_size(), entry, svfFun, pointerType);
1618 auto* arg = SVFUtil::cast<ArgValVar>(pag->getGNode(id));
1619 svfFun->addArgument(arg);
1620 // Like RetValPN, the synthetic parameter uses its function as a
1621 // diagnostic source; it does not add or replace an LLVM argument.
1622 llvmModuleSet()->addToSVFVar2LLVMValueMap(fun, arg);
1623 arg->setName("aggregate_return_" + std::to_string(arg->getArgNo()));
1624 field.second = id;
1625 }
1626 // LLVM use-list order can change on bitcode reload. Order calls by
1627 // their ICFG IDs so each call receives the same slot IDs on readback.
1628 std::vector<const CallInst*> calls;
1629 for (const User* user : fun->users())
1630 calls.push_back(SVFUtil::cast<CallInst>(user));
1631 std::sort(calls.begin(), calls.end(), [this](const CallInst* lhs,
1632 const CallInst* rhs)
1633 {
1634 return llvmModuleSet()->getCallICFGNode(lhs)->getId() <
1635 llvmModuleSet()->getCallICFGNode(rhs)->getId();
1636 });
1637 // Include calls with unused results to keep actual/formal lists aligned.
1638 for (const CallInst* call : calls)
1639 {
1640 std::vector<const Instruction*> next;
1641 LLVMUtil::getNextInsts(call, next);
1642 assert(next.size() == 1 && "ordinary call must have one continuation");
1643 AggregateCall& model = aggregateCalls[call];
1644 model.continuation = next.front();
1645 ICFGNode* callNode = llvmModuleSet()->getCallICFGNode(call);
1646 ICFGNode* resumeNode = llvmModuleSet()->getICFGNode(model.continuation);
1647 setCurrentLocation(model.continuation, model.continuation->getParent());
1648 for (const auto& field : fields)
1649 {
1650 // DummyObjVar keeps the static slot nonlocal to the callee in MemSSA.
1651 // Later extracts read the SSA snapshot, not the reused slot contents.
1652 NodeID object = pag->addDummyObjNode(pointerType);
1653 pag->getAllFieldsObjVars(object).set(object);
1654 pag->getObjTypeInfo(object)->setNumOfElements(1);
1655 pag->getObjTypeInfo(object)->setByteSizeOfObj(pointerType->getByteSize());
1656 NodeID address = pag->addDummyValNode(NodeIDAllocator::get()->allocateValueId(), callNode);
1657 NodeID value = pag->addDummyValNode(NodeIDAllocator::get()->allocateValueId(), resumeNode);
1658 model.fields.emplace(field.first, AggregateCallField{object, address, value});
1659 // Call fields use the same identity as local aggregate fields.
1660 aggregateFieldNodes.emplace(AggregateFieldKey{call, field.first}, value);
1661 // Snapshots must precede field statements at the continuation,
1662 // even when LLVM block order differs from dominance order.
1663 addLoadEdge(address, value);
1664 }
1665 }
1666 }
1667 setCurrentLocation(savedValue, savedBB);
1668}

1768void SVFIRBuilder::visitBranchInst(BranchInst &inst)
1769{
1770 NodeID brinst = getValueNode(&inst);
1771 NodeID cond;
1772 if (inst.isConditional())
1773 cond = getValueNode(inst.getCondition());
1774 else
1775 cond = pag->getNullPtr();
1776
1777 assert(inst.getNumSuccessors() <= 2 && "if/else has more than two branches?");
1778
1779 BranchStmt::SuccAndCondPairVec successors;
1780 std::vector<const Instruction*> nextInsts;
1781 LLVMUtil::getNextInsts(&inst, nextInsts);
1782 u32_t branchID = 0;
1783 for (const Instruction* succInst : nextInsts)
1784 {
1785 assert(branchID <= 1 && "if/else has more than two branches?");
1786 const ICFGNode* icfgNode = llvmModuleSet()->getICFGNode(succInst);
1787 successors.push_back(std::make_pair(icfgNode, 1-branchID));
1788 branchID++;
1789 }
1790 addBranchStmt(brinst, cond, successors);
1792 if (inst.isConditional())
1793 {
1794 for (auto& edge : llvmModuleSet()->getICFGNode(&inst)->getOutEdges())
1795 {
1796 if (IntraCFGEdge* intraEdge = SVFUtil::dyn_cast<IntraCFGEdge>(edge))
1797 {
1798 intraEdge->setConditionVar(pag->getGNode(cond));
1799 }
1800 }
1801 }

1847void SVFIRBuilder::visitSwitchInst(SwitchInst &inst)
1848{
1849 NodeID brinst = getValueNode(&inst);
1850 NodeID cond = getValueNode(inst.getCondition());
1851
1852 BranchStmt::SuccAndCondPairVec successors;
1853 std::vector<const Instruction*> nextInsts;
1854 // Resolve each LLVM successor to its first non-intrinsic instruction.
1855 LLVMUtil::getNextInsts(&inst, nextInsts);
1856
1857 // Record default first and -1 is the getSuccessorCondValue(default iter, usually 0)
1858 successors.emplace_back(llvmModuleSet()->getICFGNode(nextInsts.front()), -1);
1859
1860 // switch (x) {
1861 // case 1:
1862 // case 2: work(); break;
1863 // default: other(); break;
1864 // }
1865 // Both cases jump to the same call instruction, work(). Keep two records
1866 // for that call's ICFG node: one with case value 1, one with case value 2.
1867 // findCaseDest() returns nullptr for a block shared by multiple cases.
1868 // Read the value from each case, not from its destination block.
1869 for (const auto& switchCase : inst.cases())
1870 {
1871 const ConstantInt* condVal = switchCase.getCaseValue();
1872 s64_t val = -1;
1873 if (condVal->getBitWidth() <= 64)
1874 val = LLVMUtil::getIntegerValue(condVal).first;
1875 // In the example, both cases resolve to the same call to work().
1876 const Instruction* succInst = nextInsts[switchCase.getSuccessorIndex()];
1877 const ICFGNode* icfgNode = llvmModuleSet()->getICFGNode(succInst);
1878 successors.push_back(std::make_pair(icfgNode, val));
1879 }
1880 addBranchStmt(brinst, cond, successors);
1882 for (auto& edge : llvmModuleSet()->getICFGNode(&inst)->getOutEdges())
1883 {
1884 if (IntraCFGEdge* intraEdge = SVFUtil::dyn_cast<IntraCFGEdge>(edge))
1885 {
1886 intraEdge->setConditionVar(pag->getGNode(cond));
1887 }
1888 }
1889}

1924void SVFIRBuilder::handleDirectCall(CallBase* cs, const Function *F)
1925{
1926
1927 assert(F);
1928 CallICFGNode* callICFGNode = llvmModuleSet()->getCallICFGNode(cs);
1929 const FunObjVar* svffun = llvmModuleSet()->getFunObjVar(F);
1930 DBOUT(DPAGBuild, outs() << "handle direct call " << LLVMUtil::dumpValue(cs)
1931 << " callee " << F->getName().str() << "\n");
1932
1933 //Only handle the ret.val. if it's used as a ptr.
1934 NodeID dstrec = getValueNode(cs);
1935 //Does it actually return a ptr?
1936 if (!cs->getType()->isVoidTy())
1937 {
1938 NodeID srcret = getReturnNode(svffun);
1939 FunExitICFGNode* exitICFGNode = pag->getICFG()->getFunExitICFGNode(svffun);
1940 addRetEdge(srcret, dstrec,callICFGNode, exitICFGNode);
1941 }
1942 //Iterators for the actual and formal parameters
1943 u32_t itA = 0, ieA = cs->arg_size();
1944 Function::const_arg_iterator itF = F->arg_begin(), ieF = F->arg_end();
1945 //Go through the fixed parameters.
1946 DBOUT(DPAGBuild, outs() << " args:");
1947 for (; itF != ieF; ++itA, ++itF)
1948 {
1949 //Some programs (e.g. Linux kernel) leave unneeded parameters empty.
1950 if (itA == ieA)
1951 {
1952 DBOUT(DPAGBuild, outs() << " !! not enough args\n");
1953 break;
1954 }
1955 const Value* AA = cs->getArgOperand(itA), *FA = &*itF; //current actual/formal arg
1956
1957 DBOUT(DPAGBuild, outs() << "process actual parm "
1958 << LLVMUtil::dumpValue(AA) << "\n");
1959
1960 NodeID dstFA = getValueNode(FA);
1961 NodeID srcAA = getValueNode(AA);
1962 FunEntryICFGNode* entry = pag->getICFG()->getFunEntryICFGNode(svffun);
1963 addCallEdge(srcAA, dstFA, callICFGNode, entry);
1964 }
1965 //Any remaining actual args must be varargs.
1966 if (F->isVarArg())
1967 {
1968 NodeID vaF = getVarargNode(svffun);
1969 DBOUT(DPAGBuild, outs() << "\n varargs:");
1970 for (; itA != ieA; ++itA)
1971 {
1972 const Value* AA = cs->getArgOperand(itA);
1973 NodeID vnAA = getValueNode(AA);
1974 FunEntryICFGNode* entry = pag->getICFG()->getFunEntryICFGNode(svffun);
1975 addCallEdge(vnAA,vaF, callICFGNode,entry);
1976 }
1977 }
1978 if(itA != ieA)
1979 {
