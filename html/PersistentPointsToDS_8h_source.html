94 void clearPts(const Key &var, const Data &element) override
95 {
96 DataSet toRemoveData;
97 toRemoveData.set(element);
98 PointsToID toRemoveId = ptCache.emplacePts(toRemoveData);
99 PointsToID varId = ptsMap[var];
100 PointsToID complementId = ptCache.complementPts(varId, toRemoveId);
101 if (varId != complementId)
102 {
103 ptsMap[var] = complementId;
104 clearSingleRevPts(revPtsMap[element], var);
105 }
106 }

108 void clearFullPts(const Key& var) override
109 {
110 clearRevPts(getPts(var), var);
111 ptsMap[var] = PersistentPointsToCache<DataSet>::emptyPointsToId();
112 }

116 ptCache.remapAllPts();
117 }

120 {
121 Map<DataSet, unsigned> allPts;
122 if (liveOnly)
123 {
124 for (const typename KeyToIDMap::value_type &ki : ptsMap)
125 {
126 ++allPts[ptCache.getActualPts(ki.second)];
127 }
128 }
129 else
130 {
131 allPts = ptCache.getAllPts();
132 }

150private:
153 inline bool unionPtsFromId(const Key &dstKey, PointsToID srcId)
154 {
155 PointsToID dstId = ptsMap[dstKey];
156 PointsToID newDstId = ptCache.unionPts(dstId, srcId);
157
158 bool changed = newDstId != dstId;
159 if (changed)
160 {
161 ptsMap[dstKey] = newDstId;
162
163 // Reverse points-to only needs to be handled when dst's
164 // points-to set has changed (i.e., do it the first time only).
165 if (this->rev)
166 {
167 const DataSet &srcPts = ptCache.getActualPts(srcId);
168 for (const Data &d : srcPts) SVFUtil::insertKey(dstKey, revPtsMap[d]);
169 }
170 }
171
172 return changed;
173 }
174
175 inline void clearSingleRevPts(KeySet &revSet, const Key &k)
176 {
177 if (this->rev)
178 {
179 SVFUtil::removeKey(k, revSet);
180 }
181 }
182
183 inline void clearRevPts(const DataSet &pts, const Key &k)
184 {
185 if (this->rev)
186 {
187 for (const Data &d : pts) clearSingleRevPts(revPtsMap[d], k);
188 }
189 }
190
191protected:
192 PersistentPointsToCache<DataSet> &ptCache;
193 KeyToIDMap ptsMap;
194 RevPtsMap revPtsMap;

275 inline bool computeDiffPts(Key &var, const DataSet &all) override
276 {
277 PointsToID propaId = propaPtsMap[var];
278 PointsToID allId = ptCache.emplacePts(all);
279 // Diff is made up of the entire points-to set minus what has been propagated.
280 PointsToID diffId = ptCache.complementPts(allId, propaId);
281 diffPtsMap[var] = diffId;
282
283 // We've now propagated the entire thing.
284 propaPtsMap[var] = allId;
285
286 // Whether diff is empty or not; just need to check against the ID since it
287 // is the only empty set.
288 return diffId != ptCache.emptyPointsToId();
