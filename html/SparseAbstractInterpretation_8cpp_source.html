58void FullSparseAbstractInterpretation::joinStates(AbstractState& dst,
59 const AbstractState& src, const NodeBS& objs)
60{
61 for (NodeID obj : objs)
62 {
63 const u32_t addr = AbstractState::getVirtualMemAddress(obj);
64 const auto value = src.getLocToVal().find(src.getIDFromAddr(addr));
65 if (value != src.getLocToVal().end())
66 {
67 if (dst.getLocToVal().count(dst.getIDFromAddr(addr)))
68 dst.load(addr).join_with(value->second);
69 else
70 dst.store(addr, value->second);
71 }
72 }
73}

75bool FullSparseAbstractInterpretation::mergeStatesFromPredecessors(
76 const ICFGNode* node)
77{
78 const auto& graphEdges = objRoutes->getInputs(node);
79 AbstractState previous;
80 bool hasPrevious = false;
81 for (const SparseObjGraph::Edge* edge : graphEdges)
82 {
83 if (edge->getSrcNode()->getICFGNode() == node && hasAbsState(node))
84 {
85 previous = getAbsState(node);
86 hasPrevious = true;
87 break;
88 }
89 }
90
91 // The ICFG merge determines reachability and propagates freed addresses.
92 const bool reachable = AbstractInterpretation::mergeStatesFromPredecessors(node);
93 if (reachable)
94 {
95 AbstractState merged = getAbsState(node);
96 for (const SparseObjGraph::Edge* edge : graphEdges)
97 {
98 const SparseObjGraph::Node* sourceNode = edge->getSrcNode();
99 const ICFGNode* source = sourceNode->getICFGNode();
100 const AbstractState* sourceState = source == node
101 ? (hasPrevious ? &previous : nullptr)
102 : (hasAbsState(source)
103 ? &getAbsState(source) : nullptr);
104 if (!sourceState)
105 continue;
106
107 const AbstractState* input = sourceState;
108 AbstractState refined;
109 if (sourceNode->branch)
110 {
111 refined = *sourceState;
112 if (!isBranchEdgeFeasible(sourceNode->branch, refined) ||
113 !applyBranchRefinement(sourceNode->branch, refined))
114 continue;
115 input = &refined;
116 }
117 joinStates(merged, *input, svfir->getAllFieldsObjVars(
118 svfir->getBaseObject(sourceNode->base)));
119 }
120 updateAbsState(node, merged);
121 }
122 return reachable;
123}

125AbstractState SemiSparseAbstractInterpretation::getFullCycleHeadState(
126 const ICFGCycleWTO* cycle)
127{
128 // Start from the dense snapshot (ObjVars + any ValVars that happen to
129 // be cached at cycle_head's trace entry).
130 AbstractState snap = AbstractInterpretation::getFullCycleHeadState(cycle);
131
132 const Set<const ValVar*>& valVars = preAnalysis->getCycleValVars(cycle);
133 if (valVars.empty())
134 return snap; // no cycle ValVars known: nothing to pull
135
136 // Drop stale ValVar entries and pull each cycle ValVar from its
137 // def-site. ValVars without a genuine stored value are skipped to
138 // avoid getAbsValue's top-fallback contaminating body def-sites on
139 // the subsequent widen/narrow scatter.
140 snap.clearValVars();
141 for (const ValVar* v : valVars)
142 {
143 const ICFGNode* defSite = v->getICFGNode();
144 if (!defSite || !hasAbsValue(v, defSite))
145 continue;
146 snap[v->getId()] = getAbsValue(v, defSite);
147 }
148 return snap;
149}

151bool SemiSparseAbstractInterpretation::widenCycleState(
152 const AbstractState& prev, const AbstractState& cur, const ICFGCycleWTO* cycle)
153{
154 // Base widens, writes trace[cycle_head], and returns fixpoint bool.
155 bool fixpoint = AbstractInterpretation::widenCycleState(prev, cur, cycle);
156
157 // Scatter the widened ValVars back to their def-sites so body nodes
158 // observe the widened values on the next iteration. Matches the
159 // pre-refactor semantics: scatter unconditionally, including at
160 // widening fixpoint (see the narrowing-starts-with-stale-body issue
161 // fixed by always writing widened state back).
162 const ICFGNode* cycle_head = cycle->head()->getICFGNode();
163 const AbstractState& next = abstractTrace[cycle_head];
164 for (const auto& [id, val] : next.getVarToVal())
165 updateAbsValue(svfir->getSVFVar(id), val, cycle_head);
166 return fixpoint;
167}

169bool SemiSparseAbstractInterpretation::narrowCycleState(
170 const AbstractState& prev, const AbstractState& cur, const ICFGCycleWTO* cycle)
171{
172 // Delegate to base. It returns true on the two non-scatter cases
173 // (narrowing disabled, or narrow fixpoint); we preserve the original
174 // "skip scatter at fixpoint" semantics by bailing early here.
175 bool fixpoint = AbstractInterpretation::narrowCycleState(prev, cur, cycle);
176 if (fixpoint)
177 return true;
178
179 // Non-fixpoint: base wrote the narrowed state to trace. Scatter the
180 // narrowed ValVars back to def-sites.
181 const ICFGNode* cycle_head = cycle->head()->getICFGNode();
182 const AbstractState& next = abstractTrace[cycle_head];
183 for (const auto& [id, val] : next.getVarToVal())
184 updateAbsValue(svfir->getSVFVar(id), val, cycle_head);
185 return false;
186}

201void SemiSparseAbstractInterpretation::joinStates(AbstractState& dst,
202 const AbstractState& src)
203{
204 // ValVars live at def-sites in semi-sparse mode; they don't flow
205 // through state merges. Iterate src's ObjVar (_addrToAbsVal) entries
206 // directly and join into dst, leaving dst's ValVar map untouched.
207 // _freedAddrs (used by the null-deref detector) also rides along
208 // ICFG edges — there is no SVFG-level encoding of free events.
209 for (const auto& [id, val] : src.getLocToVal())
210 {
211 u32_t addr = AbstractState::getVirtualMemAddress(id);
212 if (dst.getLocToVal().count(id))
213 dst.load(addr).join_with(val);
214 else
215 dst.store(addr, val);
216 }
217 for (NodeID a : src.getFreedAddrs())
218 dst.addToFreedAddrs(a);
219}

221const ICFGNode* SemiSparseAbstractInterpretation::getICFGNode(
222 const ValVar* var) const
223{
224 // const ValVars are all defined in global node
225 if (!var->getICFGNode())
226 {
227 return svfir->getICFG()->getGlobalICFGNode();
228 }
229 // for return value of callsite, use the ret-site as def-site
230 else if (SVFUtil::isa<CallICFGNode>(var->getICFGNode()) &&
231 SVFUtil::isa<RetValPN>(var))
232 {
233 return SVFUtil::dyn_cast<CallICFGNode>(var->getICFGNode())
234 ->getRetICFGNode();
235 }
236 // for other ValVars, use their def-site as the node to query abstract
237 // value.
238 else
239 {
240 return var->getICFGNode();
241 }
242}

244void SemiSparseAbstractInterpretation::updateAbsValue(const ValVar* var,
245 const AbstractValue& val,
246 const ICFGNode* node)
247{
248 // Write to the var's def-site so getAbsValue stays consistent.
249 const ICFGNode* defNode = var->getICFGNode();
250 abstractTrace[defNode ? defNode : node][var->getId()] = val;
251}

254 const ValVar* var, const ICFGNode* node)
255{
256 // Read from the var's def-site (where updateAbsValue wrote it).
257 return AbstractInterpretation::getAbsValue(var, getICFGNode(var));
