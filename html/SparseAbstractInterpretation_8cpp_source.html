54void FullSparseAbstractInterpretation::joinStates(AbstractState& dst,
55 const AbstractState& src, const NodeBS& objs)
56{
57 for (NodeID obj : objs)
58 {
59 const u32_t addr = AbstractState::getVirtualMemAddress(obj);
60 const auto value = src.getLocToVal().find(src.getIDFromAddr(addr));
61 if (value != src.getLocToVal().end())
62 {
63 if (dst.getLocToVal().count(dst.getIDFromAddr(addr)))
64 dst.load(addr).join_with(value->second);
65 else
66 dst.store(addr, value->second);
67 }
68 }

71bool FullSparseAbstractInterpretation::mergeStatesFromPredecessors(
72 const ICFGNode* node)
73{
74 // An empty SVFG input is not proof of unreachability. Reuse the existing
75 // control gate and mergeFreedAddrs, not a new solver.
76 const bool reachable =
77 AbstractInterpretation::mergeStatesFromPredecessors(node);
78 if (reachable)
79 {
80 AbstractState merged = getAbsState(node);
81 for (const VFGNode* target : node->getVFGNodes())
82 {
83 for (auto it = target->InEdgeBegin(); it != target->InEdgeEnd(); ++it)
84 {
85 const VFGEdge* edge = *it;
86 const ICFGNode* source = edge->getSrcNode()->getICFGNode();
87 if (source == node)
88 {
89 // Dependencies inside one ICFG node are handled by its
90 // statement execution, not by reading its old output.
91 }
92 else if (!source || !hasAbsState(source))
93 {
94 // This direct predecessor has no available output yet.
95 }
96 else
97 {
98 // Step 1. Read this predecessor's current edge values.
99 NodeBS objs;
100 AbstractState input = CollectObjValueFlowInput(edge, objs);
101 // Step 2. JOIN the contributions of feasible, refined paths.
102 input = filterValueFlowInputByPath(edge, input);
103 // Step 3. Merge this contribution, never the whole source state.
104 joinStates(merged, input, objs);
105 }
106 }
107 }
108 updateAbsState(node, merged);
109 }
110 return reachable;
111}

113AbstractState FullSparseAbstractInterpretation::CollectObjValueFlowInput(
114 const VFGEdge* edge, NodeBS& objs)
115{
116 AbstractState input;
117 objs.clear();
118 if (const auto* indirect = SVFUtil::dyn_cast<IndirectSVFGEdge>(edge))
119 {
120 for (NodeID obj : indirect->getPointsTo())
121 objs |= svfir->getAllFieldsObjVars(svfir->getBaseObject(obj));
122 const ICFGNode* source = edge->getSrcNode()->getICFGNode();
123 joinStates(input, getAbsState(source), objs);
124 }
125 return input;
126}
127
128// Step 2: filter one SVFG input D -> N along ICFG paths.
129// D/N label the source/target ICFG nodes; B labels a branch node.
130// Snippets below are simplified C. Interval annotations describe the already
131// computed AE values, not assignments executed by this search. For narrowing
132// examples, x is a memory Obj loaded through a pointer with one known target.
133AbstractState FullSparseAbstractInterpretation::filterValueFlowInputByPath(
134 const VFGEdge* edge, const AbstractState& input)
135{
136 const ICFGNode* source = edge->getSrcNode()->getICFGNode();
137 const ICFGNode* target = edge->getDstNode()->getICFGNode();
138 NodeBS objs;
139 if (const auto* indirect = SVFUtil::dyn_cast<IndirectSVFGEdge>(edge))
140 {
141 for (NodeID obj : indirect->getPointsTo())
142 objs |= svfir->getAllFieldsObjVars(svfir->getBaseObject(obj));
143 }
144 AbstractState result;
145 if (input.getLocToVal().empty() && objs.empty())
146 {
147 // No values or queried Objs. An indirect edge still identifies its
148 // base/sub Objs even when D has not produced their values yet.
149 // this branch will return empty result
150 }
151 else if (!edge->isIntraVFGEdge() || source->getFun() != target->getFun() ||
152 !source->getFun())
153 {
154 // Call input:
155 // f(&x); // D: ActualIn in caller
156 // void f(int *p) { ... } // N: FormalIn at callee entry
157 // Call output:
158 // void f(int *p) { *p = 7; } // D: FormalOut at callee exit
159 // f(&x); // N: ActualOut at caller return point
160 // Recursive input (same function, but a non-intra SVFG edge):
161 // void f(int *p) { f(p); } // D: call ActualIn -> N: entry FormalIn
162 // Global initialization:
163 // int g = 7; // D: global node, no owning function
164 // void f() { use(g); } // N
165 // All retain D's input; no cross-procedural feasibility is proved.
166 // In one word, if edge is interprocedural, we reckon it feasible, no filter applies.
167 result = input;
168 }
169 else
170 {
171 // int x = input(); // D: x=[0,10]
172 // ...
173 // if (x > 5) { // B
174 // ...
175 // use(x); // N
176 // }
177 // Same function: search D -> B -> ... -> N, carrying x=[6,10]
178 // on B's true edge. More branches give D -> B1 -> ... -> Bk -> N.
179 // incoming's key is the iterated ICFGNode in worklist
180 // incoming's value is initialized as AbstractState pulled from SVFG predecessor
181 // the AbstractState only change some interval values at branches (branch refinement)
182 Map<const ICFGNode*, AbstractState> incoming;
183 FIFOWorkList<const ICFGNode*> worklist;
184 incoming[source] = input;
185 worklist.push(source);
186 while (!worklist.empty())
187 {
188 const ICFGNode* current = worklist.pop();
189 if (current == target)
190 {
191 // int x = read_int(); // D: x=[-10,10]
192 // if (x > 0) { // B: true edge carries x=[1,10]
193 // use(x); // N: current == target
194 // }
195 // JOIN incoming[N] into this D -> N query's result. Stop at
196 // N: do not inspect or follow any of N's outgoing edges.
197 result.joinWith(incoming[current]);
198 }
199 else
200 {
201 // int a[2] = {0,1} // D: a[0] = [0,0], a[1] = [1,1]
202 // a[1] = 2 // S: a[1] = [2,2]
203 // use(a[1]) // N: if an input carries D's old a[1],
204 // its D -> S -> N path reuses S's result when strongly updated.
205 // An input from S already has a[1]=2; S itself is not replayed.
206
207 AbstractState output = incoming[current];
208 if (current != source && hasAbsState(current))
209 {
210 const AbstractState& state = getAbsState(current);
211 for (const SVFStmt* stmt : current->getSVFStmts())
212 {
213 if (const auto* store = SVFUtil::dyn_cast<StoreStmt>(stmt))
214 {
215 const AddressValue& targets =
216 getAbsValue(store->getLHSVar(), current).getAddrs();
217 const bool strongUpdate = isStrongUpdate(targets, state);
218 for (NodeID obj : objs)
219 {
220 const u32_t addr = AbstractState::getVirtualMemAddress(obj);
221 if (targets.contains(addr) &&
222 state.getIDFromAddr(addr) == obj && state.getLocToVal().count(obj))
223 {
224 // D: a[9]=9; p=a+9; S: *p=10; N: use(a[9]).
225 // Static GEP mapping may leave D -> N, not S -> N.
226 // At S, AE identifies the carried sub-Obj a[9].
227 // Reuse S's computed value; do not execute S again.
228 if (strongUpdate)
229 output.store(addr, state.load(addr));
230 else
231 output.load(addr).join_with(state.load(addr));
232 }
233 }
234 }
235 }
236 }
237 std::vector<std::pair<const ICFGNode*, const IntraCFGEdge*>> successors;
238 if (const auto* call = SVFUtil::dyn_cast<CallICFGNode>(current))
239 {
240 const RetICFGNode* retNode = call->getRetICFGNode();
241 if (call->getCalledFunction() &&
242 SVFUtil::isExtCall(call->getCalledFunction()))
243 {
244 // D: x=0; C: a model writes x=5; N: use(x).
245 // Read the queried Objs from C's already computed
246 // output, including x=5 even when input contains x=0.
247 // C may also be D, e.g. memset(a, 7, 1) -> use(a[0]);
248 // read its model output rather than skipping the call.
249 // This is a projection of C's output, not proof that
250 // C wrote every present Obj. A joined output may be
251 // less precise than this query's path-specific values.
252 if (hasAbsState(current))
253 {
254 const AbstractState& state = getAbsState(current);
255 for (NodeID obj : objs)
256 {
257 const u32_t addr = AbstractState::getVirtualMemAddress(obj);
258 if (state.getIDFromAddr(addr) == obj && state.getLocToVal().count(obj))
259 {
260 output.store(addr, state.load(addr));
261 }
262 }
263 }
264 }
265 else
266 {
267 // Ordinary and recursive calls use the same Obj return
268 // inputs. void f(int *p) { *p=7; } f(&x); returns x=7
269 // as an Obj value. Do not execute the callee again.
270 // INTERVAL_test_16-0.c: main starts i in [0,5].
271 // void foo(int* i) {
272 // *i = *i + 1; // D: i=[1,10] at fixpoint
273 // if (*i < 10) {
274 // foo(i); // C: carried i=[1,9]
275 // // R: callee output i=[10,10]
276 // }
277 // } // N: merge both branches and return
278 // Unlike a loop exit, C -> R does not test i again.
279 // The callee's exit condition is already reflected in
280 // its Obj output. Read that output before continuing
281 // at R, without searching or executing the callee body.
282 AbstractState calleeOutput;
283 NodeBS calleeOutputObjs;
284 // R may still contain a copy of C's pre-call state.
285 // Read R's indirect SVFG predecessors, not State[R].
286 for (const VFGNode* retSVFGNode : retNode->getVFGNodes())
287 {
288 for (const VFGEdge* retEdge : retSVFGNode->getInEdges())
289 {
290 if (const auto* indirect = SVFUtil::dyn_cast<IndirectSVFGEdge>(retEdge))
291 {
292 NodeBS retObjs;
293 for (NodeID obj : indirect->getPointsTo())
294 retObjs |= svfir->getAllFieldsObjVars(svfir->getBaseObject(obj));
295 calleeOutputObjs |= retObjs;
296 const ICFGNode* calleeOutputNode = retEdge->getSrcNode()->getICFGNode();
297 if (calleeOutputNode && hasAbsState(calleeOutputNode))
298 joinStates(calleeOutput, getAbsState(calleeOutputNode), retObjs);
299 }
300 }
301 }
302 // Change only queried Objs identified by the return
303 // edges; all other carried values stay unchanged.
304 for (NodeID obj : objs)
305 {
306 if (calleeOutputObjs.test(obj))
307 {
308 const u32_t addr = AbstractState::getVirtualMemAddress(obj);
309 if (calleeOutput.getLocToVal().count(obj))
310 {
311 // Replace pre-call [1,9] with returned 10.
312 // Do not JOIN or intersect with the old value:
313 // the call may have changed this Obj.
314 output.store(addr, calleeOutput.load(addr));
315 }
316 else
317 {
318 // The return edge exists, but no Obj value
319 // is available yet. Contribute bottom until
320 // a later WTO iteration supplies the output,
321 // not the pre-call value as a guessed return.
322 output.store(addr, AbstractValue());
323 }
324 }
325 }
326 }
327 // Continue the caller-side D -> C -> R -> ... -> N path
328 // with the Obj outputs collected above. Do not enter or
329 // execute the callee again in this path query.
330 // Whether f can return is not proved here (also conservative
331 // for abort()).
332 successors.emplace_back(retNode, nullptr);
333 }
334 else
335 {
336 // int x = 1; // D
337 // int y = 2; // ordinary node: keep carried x=1
338 // x = 9; // S: still keep D's carried x=1
339 // use(x); // N: may also receive x=9 from S
340 // Follow intra edges only. Do not execute y=2 or x=9;
341 // do not infer that S kills D's contribution.
342 for (const ICFGEdge* out : current->getOutEdges())
343 {
344 if (const auto* intra = SVFUtil::dyn_cast<IntraCFGEdge>(out))
345 successors.emplace_back(intra->getDstNode(), intra);
346 }
347 }
348
349 for (const auto& [next, branch] : successors)
350 {
351 AbstractState candidate = output;
352 bool feasible = true;
353 if (!next || next->getFun() != source->getFun())
354 {
355 // Structural boundary (not a C-level branch test):
356 // D -> call f -> [missing caller return point]
357 // D[in f] -> next[in g] -> ... -> N[in f]
358 // next is missing or outside D's function. Do not
359 // extend this intra-procedural search through that hop.
360 feasible = false;
361 }
362 else if (!branch || !branch->getCondition() || !hasAbsState(current))
363 {
364 // Unconditional edge:
365 // int x = input(); // D
366 // int y = 2; // M: no branch condition
367 // use(x); // N
368 // Caller-side hop:
369 // D: ...; f(); ...; use(x); // N
370 // Branch state not available yet:
371 // D: ...; if (c) { ...; use(x); } // B -> N
372 // State[B] is absent in this iteration.
373 // Keep the input unchanged: none proves the hop false.
374 }
375 else
376 {
377 // int x = input(); // D
378 // if (x > 5) { // B: an AE state already exists
379 // use(x); // N
380 // }
381 // Read B's existing Boolean result to check its edge;
382 // narrow only this path's copy of D's input below.
383 AbstractState branchState = getAbsState(current);
384 if (!isBranchEdgeFeasible(branch, branchState))
385 // int x = 0; // D
386 // if (x > 5) { // B: condition=[0,0]
387 // use(x); // N: true edge requires 1
388 // }
389 // [0,0] cannot take the true edge: reject this hop.
390 feasible = false;
391 else
392 // int x = input(); // D: x=[0,10]
393 // if (x > 5) { // B: condition=[0,1]
394 // use(x); // N: keep true, carry x=[6,10]
395 // }
396 // Nested constraints can reject a carried input:
397 // if (x <= 5) { // B1: candidate x=[0,5]
398 // if (x > 5) { use(x); } // B2 -> N
399 // }
400 // Even if State[B2] still admits true, intersecting
401 // candidate [0,5] with [6,10] gives bottom. Other
402 // unsupported constraints conservatively keep input.
403 feasible = applyBranchRefinement(branch, candidate);
404 }
405
406 // The two feasible-arrival cases below use one diamond:
407 // int x = read_int(); // D: x=[-10,10]
408 // if (x > 0) { then_work(); } // B: true carries [1,10]
409 // else { else_work(); } // false carries [-10,0]
410 // work(); // M (= next), merge point
411 // use(x); // N
412 //
413 // D -> B
414 // true / \ false
415 // [1,10] [-10,0]
416 // \ /
417 // M -> N
418 // "First" means search arrival order, not program order.
419 // Either route can arrive first; the final JOIN is the same.
420 if (!feasible)
421 {
422 // int x = 0; // D
423 // if (x > 5) { ...; }
424 // use(x); // N is after the if
425 // Drop the true route only; the false route can still
426 // reach N. Other SVFG predecessors are not rejected.
427 }
428 else if (!incoming.count(next))
429 {
430 // Diamond above, assuming the true route arrives first:
431 // incoming[M] absent, candidate=[1,10]
432 // -> save incoming[M]=[1,10] and queue M.
433 // The false route has not contributed yet. This is
434 // the same initialization used for any first visit.
435 incoming[next] = candidate;
436 worklist.push(next);
437 }
438 else
439 {
440 // Same diamond, now the false route arrives at M:
441 // incoming[M]=[1,10], candidate=[-10,0]
442 // -> JOIN gives [-10,10]. M need not have been
443 // processed yet; its first input is already saved.
444 // Both routes started at D, not two different SVFG defs.
445 AbstractState joined = incoming[next];
446 joined.joinWith(candidate);
447 // Another repeated-arrival pattern is a loop back edge:
448 // int x = read_int(); // D: x=[-10,10]
449 // while (again()) {} // M: head / back edge
450 // use(x); // N
451 // Existing [-10,10] JOIN back-edge [-10,10] is unchanged:
452 // skip the if below and do not queue M again.
453 if (joined != incoming[next])
454 {
455 // Diamond: [1,10] -> [-10,10] enlarged M's input.
456 // Queue M so the added values can also reach N.
457 incoming[next] = std::move(joined);
458 worklist.push(next);
459 }
460 }
461 }
462 }
463 }
464 }
465 return result;
466}

468AbstractState SemiSparseAbstractInterpretation::getFullCycleHeadState(
469 const ICFGCycleWTO* cycle)
470{
471 // Start from the dense snapshot (ObjVars + any ValVars that happen to
472 // be cached at cycle_head's trace entry).
473 AbstractState snap = AbstractInterpretation::getFullCycleHeadState(cycle);
474
475 const Set<const ValVar*>& valVars = preAnalysis->getCycleValVars(cycle);
476 if (valVars.empty())
477 return snap; // no cycle ValVars known: nothing to pull
478
479 // Drop stale ValVar entries and pull each cycle ValVar from its
480 // def-site. ValVars without a genuine stored value are skipped to
481 // avoid getAbsValue's top-fallback contaminating body def-sites on
482 // the subsequent widen/narrow scatter.
483 snap.clearValVars();
484 for (const ValVar* v : valVars)
485 {
486 const ICFGNode* defSite = v->getICFGNode();
487 if (!defSite || !hasAbsValue(v, defSite))
488 continue;
489 snap[v->getId()] = getAbsValue(v, defSite);
490 }
491 return snap;
492}

494bool SemiSparseAbstractInterpretation::widenCycleState(
495 const AbstractState& prev, const AbstractState& cur, const ICFGCycleWTO* cycle)
496{
497 // Base widens, writes trace[cycle_head], and returns fixpoint bool.
498 bool fixpoint = AbstractInterpretation::widenCycleState(prev, cur, cycle);
499
500 // Scatter the widened ValVars back to their def-sites so body nodes
501 // observe the widened values on the next iteration. Matches the
502 // pre-refactor semantics: scatter unconditionally, including at
503 // widening fixpoint (see the narrowing-starts-with-stale-body issue
504 // fixed by always writing widened state back).
505 const ICFGNode* cycle_head = cycle->head()->getICFGNode();
506 const AbstractState& next = abstractTrace[cycle_head];
507 for (const auto& [id, val] : next.getVarToVal())
508 updateAbsValue(svfir->getSVFVar(id), val, cycle_head);
509 return fixpoint;
510}

512bool SemiSparseAbstractInterpretation::narrowCycleState(
513 const AbstractState& prev, const AbstractState& cur, const ICFGCycleWTO* cycle)
514{
515 // Delegate to base. It returns true on the two non-scatter cases
516 // (narrowing disabled, or narrow fixpoint); we preserve the original
517 // "skip scatter at fixpoint" semantics by bailing early here.
518 bool fixpoint = AbstractInterpretation::narrowCycleState(prev, cur, cycle);
519 if (fixpoint)
520 return true;
521
522 // Non-fixpoint: base wrote the narrowed state to trace. Scatter the
523 // narrowed ValVars back to def-sites.
524 const ICFGNode* cycle_head = cycle->head()->getICFGNode();
525 const AbstractState& next = abstractTrace[cycle_head];
526 for (const auto& [id, val] : next.getVarToVal())
527 updateAbsValue(svfir->getSVFVar(id), val, cycle_head);
528 return false;
529}

544void SemiSparseAbstractInterpretation::joinStates(AbstractState& dst,
545 const AbstractState& src)
546{
547 // ValVars live at def-sites in semi-sparse mode; they don't flow
548 // through state merges. Iterate src's ObjVar (_addrToAbsVal) entries
549 // directly and join into dst, leaving dst's ValVar map untouched.
550 // _freedAddrs (used by the null-deref detector) also rides along
551 // ICFG edges — there is no SVFG-level encoding of free events.
552 for (const auto& [id, val] : src.getLocToVal())
553 {
554 u32_t addr = AbstractState::getVirtualMemAddress(id);
555 if (dst.getLocToVal().count(id))
556 dst.load(addr).join_with(val);
557 else
558 dst.store(addr, val);
559 }
560 for (NodeID a : src.getFreedAddrs())
561 dst.addToFreedAddrs(a);
562}

564const ICFGNode* SemiSparseAbstractInterpretation::getICFGNode(
565 const ValVar* var) const
566{
567 // const ValVars are all defined in global node
568 if (!var->getICFGNode())
569 {
570 return svfir->getICFG()->getGlobalICFGNode();
571 }
572 // for return value of callsite, use the ret-site as def-site
573 else if (SVFUtil::isa<CallICFGNode>(var->getICFGNode()) &&
574 SVFUtil::isa<RetValPN>(var))
575 {
576 return SVFUtil::dyn_cast<CallICFGNode>(var->getICFGNode())
577 ->getRetICFGNode();
578 }
579 // for other ValVars, use their def-site as the node to query abstract
580 // value.
581 else
582 {
583 return var->getICFGNode();
584 }
585}
